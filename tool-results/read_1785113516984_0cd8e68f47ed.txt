    87→    87→/**
    88→    88→ * Generate the next sequential document number: FAC-2026-001
    89→    89→ */
    90→    90→export async function nextDocumentNumber(type: DocumentType, date = new Date()): Promise<string> {
    91→    91→  const year = date.getFullYear()
    92→    92→  const prefix = TYPE_PREFIXES[type]
    93→    93→
    94→    94→  // Find the highest number for this type+year
    95→    95→  const existing = await db.document.findMany({
    96→    96→    where: {
    97→    97→      type,
    98→    98→      number: { startsWith: `${prefix}-${year}-` },
    99→    99→    },
   100→   100→    orderBy: { number: 'desc' },
   101→   101→    take: 1,
   102→   102→  })
   103→   103→
   104→   104→  let next = 1
   105→   105→  if (existing.length > 0) {
   106→   106→    const match = existing[0].number.match(/-(\d+)$/)
   107→   107→    if (match) next = parseInt(match[1], 10) + 1
   108→   108→  }
   109→   109→
   110→   110→  return `${prefix}-${year}-${String(next).padStart(4, '0')}`
   111→   111→}

   115→   115→/**
   116→   116→ * Sanitize text for WinAnsi encoding (pdf-lib default Helvetica).
   117→   117→ * Replaces Unicode characters that WinAnsi cannot encode with ASCII equivalents.
   118→   118→ */
   119→   119→function sanitizeText(text: string | undefined | null): string {
   120→   120→  if (!text) return ''
   121→   121→  return String(text)
   122→   122→    // Narrow no-break space (U+202F) — used by fr-FR toLocaleString as thousands separator
   123→   123→    .replace(/\u202F/g, ' ')
   124→   124→    // No-break space (U+00A0)
   125→   125→    .replace(/\u00A0/g, ' ')
   126→   126→    // Thin space, hair space, etc.
   127→   127→    .replace(/[\u2009\u200A\u200B\u200C\u200D\u2060]/g, '')
   128→   128→    // Curly quotes → straight quotes
   129→   129→    .replace(/[\u2018\u2019]/g, "'")
   130→   130→    .replace(/[\u201C\u201D]/g, '"')
   131→   131→    // Em/en dash → hyphen
   132→   132→    .replace(/[\u2013\u2014]/g, '-')
   133→   133→    // Ellipsis → ...
   134→   134→    .replace(/\u2026/g, '...')
   135→   135→    // Bullet → *
   136→   136→    .replace(/[\u2022\u25CF]/g, '*')
   137→   137→    // Copyright, trademark
   138→   138→    .replace(/\u00A9/g, '(c)')
   139→   139→    .replace(/\u00AE/g, '(R)')
   140→   140→    .replace(/\u2122/g, '(TM)')
   141→   141→    // Euro sign is OK in WinAnsi (0x80), keep it
   142→   142→    // Strip any remaining non-WinAnsi characters
   143→   143→    .replace(/[^\x20-\x7E\x80-\xFF]/g, '?')
   144→   144→}

   146→   146→/** Safe drawText wrapper that sanitizes text for WinAnsi encoding. */
   147→   147→function drawTextSafe(
   148→   148→  page: PDFPage,
   149→   149→  text: string,
   150→   150→  options: {
   151→   151→    x: number
   152→   152→    y: number
   153→   153→    size: number
   154→   154→    font: PDFFont
   155→   155→    color?: ReturnType<typeof rgb>
   156→   156→    opacity?: number
   157→   157→    rotate?: ReturnType<typeof degrees>
   158→   158→    align?: typeof TextAlignment
   159→   159→  }
   160→   160→): void {
   161→   161→  const sanitized = sanitizeText(text)
   162→   162→  // pdf-lib doesn't support align in drawText, we handle alignment manually for totals
   163→   163→  if (options.align === TextAlignment.Right) {
   164→   164→    const textWidth = options.font.widthOfTextAtSize(sanitized, options.size)
   165→   165→    page.drawText(sanitized, {
   166→   166→      ...options,
   167→   167→      x: options.x - textWidth,
   168→   168→    })
   169→   169→  } else {
   170→   170→    page.drawText(sanitized, options)
   171→   171→  }
   172→   172→}

   174→   174→function formatMoney(amount: number, currency: string): string {
   175→   175→  const symbols: Record<string, string> = { EUR: 'EUR', USD: 'USD', GBP: 'GBP', MAD: 'MAD', AED: 'AED', SAR: 'SAR' }
   176→   176→  const symbol = symbols[currency] || currency
   177→   177→  // Use 'en-US' formatting then replace comma with dot for FR style, avoids Unicode narrow no-break space (U+202F)
   178→   178→  const formatted = Math.abs(amount).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })
   179→   179→  const result = currency === 'EUR' ? `${formatted} ${symbol}` : currency === 'USD' || currency === 'GBP' ? `${symbol} ${formatted}` : `${formatted} ${symbol}`
   180→   180→  return result
   181→   181→}

   183→   183→function formatDate(date: Date | undefined | null): string {
   184→   184→  if (!date) return '-'
   185→   185→  const d = new Date(date)
   186→   186→  const day = String(d.getDate()).padStart(2, '0')
   187→   187→  const month = String(d.getMonth() + 1).padStart(2, '0')
   188→   188→  const year = d.getFullYear()
   189→   189→  return `${day}/${month}/${year}`
   190→   190→}

   300→   300→function drawMetaBox(page: PDFPage, fontBold: PDFFont, fontRegular: PDFFont, data: DocumentData, startY: number) {
   301→   301→  const { width } = page.getSize()
   302→   302→  const boxY = startY - 10

   308→   308→    y: boxY - boxHeight,
   309→   309→    width: width - 100,
   310→   310→    height: boxHeight,
   311→   311→    color: COLORS.veryLight,
   312→   312→    borderColor: COLORS.border,
   313→   313→    borderWidth: 1,
   314→   314→  })
   315→   315→
   316→   316→  // Subject

   320→   320→  // Issue date

   324→   324→  // Due date / validity

   327→   327→    ? formatDate(new Date(Date.now() + 30 * 24 * 60 * 60 * 1000))

   329→   329→  page.drawText(dueLabel, { x: 320, y: boxY - 32, size: 9, font: fontBold, color: COLORS.gray })

   333→   333→  const statusLabel = data.type === 'receipt' ? 'Payé le :' : 'Statut :'
   334→   334→  const statusValue = data.type === 'receipt'
   335→   335→    ? formatDate(data.paidAt || new Date())
   336→   336→    : data.type === 'invoice' && data.paidAt
   337→   337→      ? `Payé le ${formatDate(data.paidAt)}`
   338→   338→      : 'En attente'
   339→   339→  page.drawText(statusLabel, { x: 64, y: boxY - 48, size: 9, font: fontBold, color: COLORS.gray })
   340→   340→  page.drawText(statusValue, { x: 160, y: boxY - 48, size: 9, font: fontBold, color: data.paidAt ? COLORS.primary : COLORS.amber })
   341→   341→
   342→   342→  return boxY - boxHeight - 20
   343→   343→}

   345→   345→function drawItemsTable(page: PDFPage, fontBold: PDFFont, fontRegular: PDFFont, data: DocumentData, startY: number): number {
   346→   346→  const { width } = page.getSize()
   347→   347→  const tableX = 50
   348→   348→  const tableWidth = width - 100
   349→   349→  let yPos = startY
   350→   350→
   351→   351→  // Table header
   352→   352→  page.drawRectangle({
   353→   353→    x: tableX,
   354→   354→    y: yPos - 22,
   355→   355→    width: tableWidth,
   356→   356→    height: 22,
   357→   357→    color: COLORS.primary,
   358→   358→  })
   359→   359→
   360→   360→  page.drawText('Description', { x: tableX + 12, y: yPos - 15, size: 9, font: fontBold, color: COLORS.white })
   361→   361→  page.drawText('Qté', { x: tableX + tableWidth - 230, y: yPos - 15, size: 9, font: fontBold, color: COLORS.white })
   362→   362→  page.drawText('Prix unitaire', { x: tableX + tableWidth - 170, y: yPos - 15, size: 9, font: fontBold, color: COLORS.white })
   363→   363→  page.drawText('Total', { x: tableX + tableWidth - 70, y: yPos - 15, size: 9, font: fontBold, color: COLORS.white })
   364→   364→
   365→   365→  yPos -= 22
   366→   366→
   367→   367→  // Items rows
   368→   368→  let altRow = false
   369→   369→  let subtotal = 0
   370→   370→  for (const item of data.items) {
   371→   371→    const rowHeight = 28
   372→   372→    if (altRow) {
   373→   373→      page.drawRectangle({
   374→   374→        x: tableX,
   375→   375→        y: yPos - rowHeight,
   376→   376→        width: tableWidth,
   377→   377→        height: rowHeight,
   378→   378→        color: COLORS.veryLight,
   379→   379→      })
   380→   380→    }
   381→   381→    altRow = !altRow
   382→   382→
   383→   383→    // Description (truncated if too long)
   384→   384→    const desc = item.description.length > 65 ? item.description.slice(0, 62) + '...' : item.description
   385→   385→    page.drawText(desc, { x: tableX + 12, y: yPos - 18, size: 9, font: fontRegular, color: COLORS.dark })
   386→   386→
   387→   387→    page.drawText(String(item.quantity), { x: tableX + tableWidth - 230, y: yPos - 18, size: 9, font: fontRegular, color: COLORS.dark })
   388→   388→    page.drawText(formatMoney(item.unitPrice, data.currency), { x: tableX + tableWidth - 170, y: yPos - 18, size: 9, font: fontRegular, color: COLORS.dark })
   389→   389→    page.drawText(formatMoney(item.total, data.currency), { x: tableX + tableWidth - 70, y: yPos - 18, size: 9, font: fontBold, color: COLORS.dark })
   390→   390→
   391→   391→    subtotal += item.total
   392→   392→    yPos -= rowHeight
   393→   393→  }
   394→   394→
   395→   395→  // Border around table
   396→   396→  page.drawRectangle({
   397→   397→    x: tableX,
   398→   398→    y: yPos,
   399→   399→    width: tableWidth,
   400→   400→    height: startY - yPos,
   401→   401→    borderColor: COLORS.border,
   402→   402→    borderWidth: 1,
   403→   403→  })
   404→   404→
   405→   405→  // Totals
   406→   406→  const taxRate = data.taxRate || 0
   407→   407→  const taxAmount = subtotal * (taxRate / 100)
   408→   408→  const total = subtotal + taxAmount
   409→   409→
   410→   410→  yPos -= 20
   411→   411→  const totalsX = tableX + tableWidth - 240
   412→   412→
   413→   413→  // Subtotal
   414→   414→  page.drawText('Sous-total :', { x: totalsX, y: yPos, size: 10, font: fontRegular, color: COLORS.gray })
   415→   415→  page.drawText(formatMoney(subtotal, data.currency), { x: tableX + tableWidth - 12 - 90, y: yPos, size: 10, font: fontRegular, color: COLORS.dark, align: TextAlignment.Right })
   416→   416→
   417→   417→  yPos -= 18
   418→   418→  if (taxRate > 0) {
   419→   419→    page.drawText(`TVA (${taxRate}%) :`, { x: totalsX, y: yPos, size: 10, font: fontRegular, color: COLORS.gray })
   420→   420→    page.drawText(formatMoney(taxAmount, data.currency), { x: tableX + tableWidth - 12 - 90, y: yPos, size: 10, font: fontRegular, color: COLORS.dark, align: TextAlignment.Right })
   421→   421→    yPos -= 18
   422→   422→  }
   423→   423→
   424→   424→  // Total (highlighted)
   425→   425→  page.drawRectangle({
   426→   426→    x: totalsX - 12,
   427→   427→    y: yPos - 6,
   428→   428→    width: tableX + tableWidth - totalsX,
   429→   429→    height: 26,
   430→   430→    color: COLORS.primaryLight,
   431→   431→  })
   432→   432→  page.drawText('TOTAL :', { x: totalsX, y: yPos, size: 12, font: fontBold, color: COLORS.primaryDark })
   433→   433→  page.drawText(formatMoney(total, data.currency), { x: tableX + tableWidth - 12 - 90, y: yPos, size: 12, font: fontBold, color: COLORS.primaryDark, align: TextAlignment.Right })
   434→   434→
   435→   435→  return yPos - 40
   436→   436→}

   497→   497→async function buildPdf(rawData: DocumentData): Promise<Uint8Array> {
   498→   498→  // Sanitize user-provided data for PDF metadata (setTitle, setSubject, etc.)
   499→   499→  const data: DocumentData = {
   500→   500→    ...rawData,
   501→   501→    recipientName: sanitizeText(rawData.recipientName),
   502→   502→    recipientEmail: sanitizeText(rawData.recipientEmail),
   503→   503→    recipientCompany: sanitizeText(rawData.recipientCompany),
   504→   504→    recipientAddress: sanitizeText(rawData.recipientAddress),
   505→   505→    recipientCountry: sanitizeText(rawData.recipientCountry),
   506→   506→    subject: sanitizeText(rawData.subject),
   507→   507→    notes: sanitizeText(rawData.notes),
   508→   508→    agreementTerms: sanitizeText(rawData.agreementTerms),
   509→   509→    items: rawData.items.map((item) => ({
   510→   510→      ...item,
   511→   511→      description: sanitizeText(item.description),
   512→   512→    })),
   513→   513→  }
   514→   514→
   515→   515→  const pdfDoc = await PDFDocument.create()
   516→   516→  const fontRegular = await pdfDoc.embedFont(StandardFonts.Helvetica)
   517→   517→  const fontBold = await pdfDoc.embedFont(StandardFonts.HelveticaBold)
   518→   518→  const fontItalic = await pdfDoc.embedFont(StandardFonts.HelveticaOblique)
   519→   519→
   520→   520→  pdfDoc.setTitle(sanitizeText(`${TYPE_META[data.type].label} ${data.number} - HireNova`))
   521→   521→  pdfDoc.setAuthor(sanitizeText('HireNova - E-Society 2050'))
   522→   522→  pdfDoc.setSubject(data.subject)
   523→   523→  pdfDoc.setCreator('HireNova Document Engine')
   524→   524→  pdfDoc.setProducer('HireNova Auto-Generator')
   525→   525→  pdfDoc.setCreationDate(new Date())
   526→   526→
   527→   527→  const page = pdfDoc.addPage([595.28, 841.89]) // A4
   528→   528→
   529→   529→  // CRITICAL: Override drawText to auto-sanitize ALL text for WinAnsi encoding.
   530→   530→  // This prevents "WinAnsi cannot encode" errors with Unicode chars (em dash, narrow no-break space, accented chars, etc.)
   531→   531→  // Also handles right-alignment manually since pdf-lib's drawText doesn't support align natively.
   532→   532→  const originalDrawText = page.drawText.bind(page)
   533→   533→  ;(page as any).drawText = (text: string, options: any) => {
   534→   534→    const sanitized = sanitizeText(text)
   535→   535→    if (options.align === TextAlignment.Right) {
   536→   536→      const textWidth = options.font.widthOfTextAtSize(sanitized, options.size)
   537→   537→      const { align, ...rest } = options
   538→   538→      return originalDrawText(sanitized, { ...rest, x: options.x - textWidth })
   539→   539→    }
   540→   540→    const { align, ...rest } = options
   541→   541→    return originalDrawText(sanitized, rest)
   542→   542→  }
   543→   543→
   544→   544→  const meta = TYPE_META[data.type]
   545→   545→  let yPos = drawHeader(page, fontBold, fontRegular, data, meta.label, meta.color)
   546→   546→  yPos = drawMetaBox(page, fontBold, fontRegular, data, yPos)
   547→   547→
   548→   548→  // For agreement type, draw contract clauses instead of items table
   549→   549→  if (data.type === 'agreement' && data.agreementTerms) {
   550→   550→    yPos = drawAgreementClauses(page, fontBold, fontRegular, fontItalic, data, yPos)
   551→   551→  } else {
   552→   552→    yPos = drawItemsTable(page, fontBold, fontRegular, data, yPos)
   553→   553→  }
   554→   554→
   555→   555→  // Payment terms / acceptance block
   556→   556→  if (data.type === 'quote') {
   557→   557→    yPos = drawAcceptanceBlock(page, fontBold, fontRegular, yPos)
   558→   558→  } else if (data.type === 'invoice') {
   559→   559→    yPos = drawPaymentTerms(page, fontBold, fontRegular, data, yPos)
   560→   560→  }
   561→   561→
   562→   562→  drawFooter(page, fontBold, fontRegular, data)
   563→   563→
   564→   564→  const pdfBytes = await pdfDoc.save()
   565→   565→  return pdfBytes
   566→   566→}

   568→   568→function drawAgreementClauses(
   569→   569→  page: PDFPage,
   570→   570→  fontBold: PDFFont,
   571→   571→  fontRegular: PDFFont,
   572→   572→  fontItalic: PDFFont,
   573→   573→  data: DocumentData,

   576→   576→  let yPos = startY
   577→   577→
   578→   578→  page.drawText('CLAUSES DU CONTRAT', { x: 50, y: yPos, size: 11, font: fontBold, color: COLORS.primaryDark })
   579→   579→  yPos -= 20
   580→   580→
   581→   581→  const terms = data.agreementTerms || ''
   582→   582→  const clauses = terms.split('\n\n').filter(Boolean)
   583→   583→
   584→   584→  for (let i = 0; i < clauses.length; i++) {
   585→   585→    const clause = clauses[i]
   586→   586→    const title = `Article ${i + 1}`
   587→   587→    page.drawText(title, { x: 50, y: yPos, size: 10, font: fontBold, color: COLORS.dark })
   588→   588→    yPos -= 14
   589→   589→
   590→   590→    // Wrap text

   592→   592→    const words = clause.split(' ')
   593→   593→    let line = ''
   594→   594→    const lines: string[] = []
   595→   595→    for (const word of words) {
   596→   596→      const testLine = line ? line + ' ' + word : word
   597→   597→      try {
   598→   598→        const testWidth = fontRegular.widthOfTextAtSize(testLine, 9)
   599→   599→        if (testWidth > maxWidth) {
   600→   600→          if (line) lines.push(line)
   601→   601→          line = word
   602→   602→        } else {
   603→   603→          line = testLine
   604→   604→        }
   605→   605→      } catch {
   606→   606→        line = line ? line + ' ' + word : word
   607→   607→      }
   608→   608→    }
   609→   609→    if (line) lines.push(line)
   610→   610→
   611→   611→    for (const ln of lines.slice(0, 6)) {
   612→   612→      page.drawText(ln.slice(0, 100), { x: 60, y: yPos, size: 9, font: fontRegular, color: COLORS.gray })

   618→   618→
   619→   619→  // Signatures
   620→   620→  yPos = Math.max(yPos, 200)
   621→   621→  page.drawText('Signatures', { x: 50, y: yPos, size: 11, font: fontBold, color: COLORS.dark })
   622→   622→  yPos -= 20
   623→   623→
   624→   624→  page.drawLine({ start: { x: 50, y: yPos }, end: { x: 230, y: yPos }, thickness: 1, color: COLORS.gray })

   626→   626→
   627→   627→  page.drawText('Pour HireNova', { x: 50, y: yPos - 14, size: 9, font: fontBold, color: COLORS.dark })
   628→   628→  page.drawText(data.recipientName, { x: 320, y: yPos - 14, size: 9, font: fontBold, color: COLORS.dark })
   629→   629→
   630→   630→  page.drawText('Date : _______________', { x: 50, y: yPos - 30, size: 9, font: fontRegular, color: COLORS.gray })
   631→   631→  page.drawText('Date : _______________', { x: 320, y: yPos - 30, size: 9, font: fontRegular, color: COLORS.gray })
