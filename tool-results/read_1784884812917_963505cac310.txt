    41→

    53→]
    54→
    55→export default function CVForm() {
    56→  const {
    57→    formStep,
    58→    setFormStep,
    59→    language,
    60→    setLanguage,
    61→    template,
    62→    setTemplate,
    63→    formData,
    64→    updateFormData,
    65→    setStep,
    66→    setIsGenerating,
    67→    setGeneratedCV,
    68→    setError,
    69→    updateCLFormData,
    70→    setGeneratedCL,
    71→    setIsCLGenerating,
    72→    setCLError,
    73→  } = useCVStore()
    74→
    75→  const [isSubmitting, setIsSubmitting] = useState(false)
    76→  const [phoneCountryCode, setPhoneCountryCode] = useState<string>('')
    77→
    78→  function handlePhoneCountryChange(code: string) {
    79→    setPhoneCountryCode(code)
    80→    const country = countries.find((c) => c.code === code)
    81→    if (country) {
    82→      // Remove any existing dial code from phone, prepend new one
    83→      const currentNumber = formData.phone.replace(/^[+\d\s\-()]+/, '').trim()
    84→      updateFormData({ phone: currentNumber ? `${country.dial} ${currentNumber}` : country.dial })
    85→    }
    86→  }
    87→
    88→  function handlePhoneNumberChange(number: string) {
    89→    const country = countries.find((c) => c.code === phoneCountryCode)
    90→    const dial = country?.dial || ''
    91→    const cleaned = number.replace(/[^\d\s]/g, '').trim()
    92→    updateFormData({ phone: dial ? `${dial} ${cleaned}` : cleaned })
    93→  }
    94→
    95→  function getPhoneNumberOnly(): string {
    96→    const country = countries.find((c) => c.code === phoneCountryCode)
    97→    const dial = country?.dial || ''
    98→    return formData.phone.startsWith(dial) ? formData.phone.slice(dial.length).trim() : formData.phone.replace(/^[+\d\s\-()]+/, '').trim()
    99→  }
   100→
   101→  const slideVariants = {
   102→    enter: (direction: number) => ({ x: direction > 0 ? 300 : -300, opacity: 0 }),
   103→    center: { x: 0, opacity: 1 },
   104→    exit: (direction: number) => ({ x: direction < 0 ? 300 : -300, opacity: 0 }),
   105→  }
   106→
   107→  function validateStep(step: number): boolean {
   108→    if (step === 0) {
   109→      if (!formData.fullName.trim() || !formData.email.trim()) {
   110→        toast.error(t(language, 'errorFillFields'))
   111→        return false
   112→      }
   113→    }
   114→    if (step === 1) {
   115→      if (!formData.targetJob.trim()) {
   116→        toast.error(t(language, 'errorFillFields'))
   117→        return false
   118→      }
   119→    }
   120→    return true
   121→  }
   122→
   123→  function handleNext() {
   124→    if (!validateStep(formStep)) return
   125→    if (formStep < 3) setFormStep(formStep + 1)
   126→  }
   127→
   128→  function handlePrevious() {
   129→    if (formStep > 0) setFormStep(formStep - 1)
   130→  }
   131→
   132→  async function handleGenerate() {
   133→    if (!validateStep(formStep)) return
   134→    setIsSubmitting(true)
   135→    setStep('generating')

   137→    setError(null)
   138→    setIsCLGenerating(formData.companyName.trim().length > 0)
   139→    setCLError(null)
   140→
   141→    const hasCompany = formData.companyName.trim().length > 0
   142→
   143→    try {
   144→      // Generate CV
   145→      const res = await fetch('/api/generate-cv', {
   146→        method: 'POST',
   147→        headers: { 'Content-Type': 'application/json' },
   148→        body: JSON.stringify({ ...formData, language }),
   149→      })
   150→
   151→      const data = await res.json()

   153→        if (data.code === 'LIMIT_REACHED') {
   154→          setStep('landing')
   155→          toast.error(data.error, { duration: 6000 })
   156→          return
   157→        }
   158→        throw new Error(data.error || 'Erreur lors de la génération')
   159→      }
   160→
   161→      const generatedCV = data.cv
   162→      setGeneratedCV(generatedCV)
   163→
   164→      // If company name provided, auto-generate cover letter simultaneously
   165→      if (hasCompany) {
   166→        try {
   167→          // Auto-sync personal info to CL form data
   168→          updateCLFormData({
   169→            fullName: formData.fullName,
   170→            email: formData.email,
   171→            phone: formData.phone,
   172→            address: formData.address,
   173→            location: formData.location,
   174→            companyName: formData.companyName,
   175→            hiringManager: formData.hiringManager,
   176→            jobTitle: formData.targetJob,
   177→            tone: formData.clTone,
   178→          })
   179→
   180→          const clRes = await fetch('/api/generate-cover-letter', {
   181→            method: 'POST',
   182→            headers: { 'Content-Type': 'application/json' },
   183→            body: JSON.stringify({
   184→              fullName: formData.fullName,
   185→              email: formData.email,
   186→              phone: formData.phone,
   187→              address: formData.address,
   188→              location: formData.location,
   189→              companyName: formData.companyName,
   190→              hiringManager: formData.hiringManager,
   191→              jobTitle: formData.targetJob,
   192→              tone: formData.clTone,
   193→              language,
   194→              generatedCVSummary: generatedCV.summary,
   195→              generatedCVExperience: generatedCV.experience,
   196→              generatedCVEducation: generatedCV.education,
   197→              generatedCVSkills: generatedCV.skills,
   198→              generatedCVLanguages: generatedCV.languages,
   199→            }),
   200→          })
   201→
   202→          const clData = await clRes.json()
   203→          if (clRes.ok && clData.letter) {
   204→            setGeneratedCL(clData.letter)
   205→          }
   206→        } catch {
   207→          // CL generation failed silently - CV is still valid
   208→          setIsCLGenerating(false)
   209→        }
   210→      }
   211→
   212→      setTimeout(() => {
   213→        setStep('preview')
   214→        setIsGenerating(false)
   215→        setIsCLGenerating(false)

   219→      setIsGenerating(false)
   220→      setIsCLGenerating(false)
   221→      setStep('form')
   222→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue')

   224→      setIsSubmitting(false)
   225→    }
   226→  }

   232→      {/* Header */}
   233→      <header className="w-full px-4 sm:px-6 lg:px-8 py-4 border-b bg-white/80 backdrop-blur-sm sticky top-0 z-50">
   234→        <div className="max-w-3xl mx-auto flex items-center justify-between">
   235→          <button
   236→            onClick={() => setStep('landing')}
   237→            className="flex items-center gap-2 text-muted-foreground hover:text-foreground transition-colors"

   239→            <div className="w-8 h-8 rounded-lg bg-emerald-600 flex items-center justify-center">
   240→              <FileText className="w-4 h-4 text-white" />
   241→            </div>

   252→                    : 'text-muted-foreground hover:text-foreground'

   279→                      !isCompleted && i !== formStep ? 'opacity-40' : ''

   281→                  >
   282→                    <div
   283→                      className={`w-10 h-10 rounded-full flex items-center justify-center transition-all ${
   284→                        isActive
   285→                          ? 'bg-emerald-600 text-white shadow-lg shadow-emerald-600/25'
   286→                          : isCompleted
   287→                            ? 'bg-emerald-100 text-emerald-700'
   288→                            : 'bg-muted text-muted-foreground'
   289→                      }`}
   290→                    >
   291→                      <Icon className="w-5 h-5" />
   292→                    </div>
   293→                    <span
   294→                      className={`text-xs font-medium text-center hidden sm:block ${
   295→                        isActive ? 'text-emerald-600' : 'text-muted-foreground'
   296→                      }`}
   297→                    >
   298→                      {t(language, titleKey)}
   299→                    </span>
   300→                  </button>
   301→                )
   302→              })}
   303→            </div>
   304→            {/* Progress bar */}
   305→            <div className="h-1.5 bg-muted rounded-full overflow-hidden">
   306→              <motion.div
   307→                className="h-full bg-emerald-500 rounded-full"
   308→                animate={{ width: `${((formStep + 1) / 4) * 100}%` }}
   309→                transition={{ duration: 0.4, ease: 'easeOut' }}
   310→              />
   311→            </div>
   312→          </div>
   313→
   314→          {/* Form Content */}
   315→          <div className="bg-white rounded-2xl border shadow-sm p-6 sm:p-8 min-h-[400px] relative overflow-hidden">
   316→            <AnimatePresence mode="wait" custom={1}>
   317→              <motion.div
   318→                key={formStep}
   319→                custom={1}
   320→                variants={slideVariants}
   321→                initial="enter"
   322→                animate="center"
   323→                exit="exit"
   324→                transition={{ duration: 0.3, ease: 'easeInOut' }}
   325→              >
   326→                <h2 className="text-xl font-bold text-foreground mb-6 flex items-center gap-2">
   327→                  {(() => {
   328→                    const Icon = stepIcons[formStep]
   329→                    return <Icon className="w-5 h-5 text-emerald-600" />
   330→                  })()}
   331→                  {t(language, stepTitles[formStep])}
   332→                </h2>
   333→
   334→                {/* Step 1: Personal Info */}
   335→                {formStep === 0 && (
   336→                  <div className="space-y-4">
   337→                    {/* Photo Upload */}
   338→                    <div>
   339→                      <Label>{t(language, 'photo')}</Label>
   340→                      <div className="mt-1.5">
   341→                        {formData.photo ? (
   342→                          <div className="relative inline-block">
   343→                            <img
   344→                              src={formData.photo}
   345→                              alt="Photo"
   346→                              className="w-24 h-24 rounded-full object-cover border-2 border-emerald-200"
   347→                            />
   348→                            <button
   349→                              type="button"
   350→                              onClick={() => updateFormData({ photo: '' })}
   351→                              className="absolute -top-1 -right-1 w-6 h-6 rounded-full bg-red-500 text-white flex items-center justify-center hover:bg-red-600 transition-colors cursor-pointer shadow-sm"
   352→                            >
   353→                              <X className="w-3 h-3" />
   354→                            </button>
   355→                          </div>
   356→                        ) : (
   357→                          <label className="flex flex-col items-center justify-center w-full h-32 border-2 border-dashed border-muted-foreground/25 rounded-xl cursor-pointer hover:border-emerald-400 hover:bg-emerald-50/50 transition-all group">
   358→                            <input
   359→                              type="file"
   360→                              accept="image/*"
   361→                              className="hidden"
   362→                              onChange={(e) => {
   363→                                const file = e.target.files?.[0]
   364→                                if (!file) return
   365→                                if (file.size > 5 * 1024 * 1024) {
   366→                                  toast.error(language === 'fr' ? 'Image trop volumineuse (max 5 Mo)' : language === 'en' ? 'Image too large (max 5 MB)' : 'الصورة كبيرة جداً (الحد الأقصى 5 ميغابايت)')
   367→                                  return
   368→                                }
   369→                                const reader = new FileReader()
   370→                                reader.onload = () => {
   371→                                  updateFormData({ photo: reader.result as string })
   372→                                }
   373→                                reader.readAsDataURL(file)
   374→                              }}
   375→                            />
   376→                            <Camera className="w-8 h-8 text-muted-foreground/40 group-hover:text-emerald-500 transition-colors" />
   377→                            <span className="text-xs text-muted-foreground/60 mt-1.5 group-hover:text-emerald-600 transition-colors">
   378→                              {t(language, 'photoPlaceholder')}
   379→                            </span>
   380→                          </label>
   381→                        )}
   382→                      </div>
   383→                      <div className="flex items-start gap-1.5 mt-2">
   384→                        <Info className="w-3.5 h-3.5 text-amber-500 shrink-0 mt-0.5" />
   385→                        <p className="text-xs text-muted-foreground">{t(language, 'photoNote')}</p>
   386→                      </div>
   387→                    </div>
   388→                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   389→                      <div className="sm:col-span-2">
   390→                        <Label htmlFor="fullName">{t(language, 'fullName')} *</Label>
   391→                        <Input
   392→                          id="fullName"
   393→                          value={formData.fullName}
   394→                          onChange={(e) => updateFormData({ fullName: e.target.value })}
   395→                          placeholder={language === 'fr' ? 'Jean Dupont' : language === 'en' ? 'John Smith' : 'محمد أحمد'}
   396→                          className="mt-1.5"
   397→                        />
   398→                      </div>
   399→                      <div>
   400→                        <Label htmlFor="email">{t(language, 'email')} *</Label>
   401→                        <Input
   402→                          id="email"
   403→                          type="email"
   404→                          value={formData.email}
   405→                          onChange={(e) => updateFormData({ email: e.target.value })}
   406→                          placeholder="jean@exemple.com"
   407→                          className="mt-1.5"
   408→                        />
   409→                      </div>
   410→                      <div className="sm:col-span-2">
   411→                        <Label>{t(language, 'phone')}</Label>
   412→                        <div className="mt-1.5 flex gap-2">
   413→                          <select
   414→                            value={phoneCountryCode}
   415→                            onChange={(e) => handlePhoneCountryChange(e.target.value)}
   416→                            className="h-9 w-[180px] shrink-0 rounded-md border border-input bg-background px-3 text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500 focus:ring-offset-1 appearance-none cursor-pointer"
   417→                            aria-label={t(language, 'phoneCountry')}
   418→                          >
   419→                            <option value="">{t(language, 'phoneCountryPlaceholder')}</option>
   420→                            {countries.map((c) => (
   421→                              <option key={c.code} value={c.code}>
   422→                                {c.flag} {c.dial} {c.name}
   423→                              </option>
   424→                            ))}
   425→                          </select>
   426→                          <Input
   427→                            type="tel"
   428→                            value={getPhoneNumberOnly()}
   429→                            onChange={(e) => handlePhoneNumberChange(e.target.value)}
   430→                            placeholder={language === 'fr' ? '600 000 000' : language === 'en' ? '600 000 000' : language === 'ar' ? '600 000 000' : '600 000 000'}
   431→                            className="flex-1"
   432→                          />
   433→                        </div>
   434→                      </div>
   435→                      <div>
   436→                        <Label htmlFor="address">{t(language, 'address')}</Label>
   437→                        <Input
   438→                          id="address"
   439→                          value={formData.address}
   440→                          onChange={(e) => updateFormData({ address: e.target.value })}
   441→                          placeholder={language === 'fr' ? '123 Rue Mohammed V' : language === 'en' ? '123 Main Street' : language === 'ar' ? 'شارع محمد الخامس 123' : 'Calle Ejemplo 123'}
   442→                          className="mt-1.5"
   443→                        />
   444→                      </div>
   445→                      <div>
   446→                        <Label htmlFor="location">{t(language, 'location')}</Label>
   447→                        <Input
   448→                          id="location"
   449→                          value={formData.location}
   450→                          onChange={(e) => updateFormData({ location: e.target.value })}
   451→                          placeholder={language === 'fr' ? 'Casablanca, Maroc' : language === 'en' ? 'London, UK' : language === 'ar' ? 'الدار البيضاء، المغرب' : 'Madrid, España'}
   452→                          className="mt-1.5"

   471→                          placeholder="www.monsite.com"
   472→                          className="mt-1.5"
   473→                        />
   474→                      </div>
   475→                    </div>
   476→
   477→                    {/* Birth Information */}
   478→                    <div className="mt-6 pt-5 border-t border-stone-100">

   480→                        <CalendarDays className="w-4 h-4 text-emerald-600" />
   481→                        {t(language, 'personalInfo')}
   482→                      </Label>
   483→                      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">

   485→                          <Label htmlFor="dateOfBirth">{t(language, 'dateOfBirth')}</Label>
   486→                          <Input
   487→                            id="dateOfBirth"
   488→                            type="date"
   489→                            value={formData.dateOfBirth}
   490→                            onChange={(e) => updateFormData({ dateOfBirth: e.target.value })}
   491→                            className="mt-1.5"
   492→                          />
   493→                        </div>
   494→                        <div>
   495→                          <Label htmlFor="birthPlace">{t(language, 'birthPlace')}</Label>
   496→                          <Input
   497→                            id="birthPlace"
   498→                            value={formData.birthPlace}

   500→                            placeholder={language === 'fr' ? 'Casablanca' : language === 'en' ? 'Casablanca' : language === 'ar' ? 'الدار البيضاء' : 'Madrid'}

   518→
   519→                {/* Step 2: Career Goals & Cover Letter */}
   520→                {formStep === 1 && (
