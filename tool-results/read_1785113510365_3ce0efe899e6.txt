    87→/**
    88→ * Generate the next sequential document number: FAC-2026-001
    89→ */
    90→export async function nextDocumentNumber(type: DocumentType, date = new Date()): Promise<string> {
    91→  const year = date.getFullYear()
    92→  const prefix = TYPE_PREFIXES[type]
    93→
    94→  // Find the highest number for this type+year
    95→  const existing = await db.document.findMany({
    96→    where: {
    97→      type,
    98→      number: { startsWith: `${prefix}-${year}-` },
    99→    },
   100→    orderBy: { number: 'desc' },
   101→    take: 1,
   102→  })
   103→
   104→  let next = 1
   105→  if (existing.length > 0) {
   106→    const match = existing[0].number.match(/-(\d+)$/)
   107→    if (match) next = parseInt(match[1], 10) + 1
   108→  }
   109→
   110→  return `${prefix}-${year}-${String(next).padStart(4, '0')}`
   111→}

   115→/**
   116→ * Sanitize text for WinAnsi encoding (pdf-lib default Helvetica).
   117→ * Replaces Unicode characters that WinAnsi cannot encode with ASCII equivalents.
   118→ */
   119→function sanitizeText(text: string | undefined | null): string {
   120→  if (!text) return ''
   121→  return String(text)
   122→    // Narrow no-break space (U+202F) — used by fr-FR toLocaleString as thousands separator
   123→    .replace(/\u202F/g, ' ')
   124→    // No-break space (U+00A0)
   125→    .replace(/\u00A0/g, ' ')
   126→    // Thin space, hair space, etc.
   127→    .replace(/[\u2009\u200A\u200B\u200C\u200D\u2060]/g, '')
   128→    // Curly quotes → straight quotes
   129→    .replace(/[\u2018\u2019]/g, "'")
   130→    .replace(/[\u201C\u201D]/g, '"')
   131→    // Em/en dash → hyphen
   132→    .replace(/[\u2013\u2014]/g, '-')
   133→    // Ellipsis → ...
   134→    .replace(/\u2026/g, '...')
   135→    // Bullet → *
   136→    .replace(/[\u2022\u25CF]/g, '*')
   137→    // Copyright, trademark
   138→    .replace(/\u00A9/g, '(c)')
   139→    .replace(/\u00AE/g, '(R)')
   140→    .replace(/\u2122/g, '(TM)')
   141→    // Euro sign is OK in WinAnsi (0x80), keep it
   142→    // Strip any remaining non-WinAnsi characters
   143→    .replace(/[^\x20-\x7E\x80-\xFF]/g, '?')
   144→}

   146→/** Safe drawText wrapper that sanitizes text for WinAnsi encoding. */
   147→function drawTextSafe(
   148→  page: PDFPage,
   149→  text: string,
   150→  options: {
   151→    x: number
   152→    y: number
   153→    size: number
   154→    font: PDFFont
   155→    color?: ReturnType<typeof rgb>
   156→    opacity?: number
   157→    rotate?: ReturnType<typeof degrees>
   158→    align?: typeof TextAlignment
   159→  }
   160→): void {
   161→  const sanitized = sanitizeText(text)
   162→  // pdf-lib doesn't support align in drawText, we handle alignment manually for totals
   163→  if (options.align === TextAlignment.Right) {
   164→    const textWidth = options.font.widthOfTextAtSize(sanitized, options.size)
   165→    page.drawText(sanitized, {
   166→      ...options,
   167→      x: options.x - textWidth,
   168→    })
   169→  } else {
   170→    page.drawText(sanitized, options)
   171→  }
   172→}

   174→function formatMoney(amount: number, currency: string): string {
   175→  const symbols: Record<string, string> = { EUR: 'EUR', USD: 'USD', GBP: 'GBP', MAD: 'MAD', AED: 'AED', SAR: 'SAR' }
   176→  const symbol = symbols[currency] || currency
   177→  // Use 'en-US' formatting then replace comma with dot for FR style, avoids Unicode narrow no-break space (U+202F)
   178→  const formatted = Math.abs(amount).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })
   179→  const result = currency === 'EUR' ? `${formatted} ${symbol}` : currency === 'USD' || currency === 'GBP' ? `${symbol} ${formatted}` : `${formatted} ${symbol}`
   180→  return result
   181→}

   183→function formatDate(date: Date | undefined | null): string {
   184→  if (!date) return '-'
   185→  const d = new Date(date)
   186→  const day = String(d.getDate()).padStart(2, '0')
   187→  const month = String(d.getMonth() + 1).padStart(2, '0')
   188→  const year = d.getFullYear()
   189→  return `${day}/${month}/${year}`
   190→}

   301→  const { width } = page.getSize()
   302→  const boxY = startY - 10

   309→    width: width - 100,
   310→    height: boxHeight,
   311→    color: COLORS.veryLight,
   312→    borderColor: COLORS.border,
   313→    borderWidth: 1,
   314→  })

   317→  page.drawText('Objet :', { x: 64, y: boxY - 16, size: 9, font: fontBold, color: COLORS.gray })

   322→  page.drawText(formatDate(data.issueDate || new Date()), { x: 160, y: boxY - 32, size: 9, font: fontRegular, color: COLORS.dark })
   323→
   324→  // Due date / validity
   325→  const dueLabel = data.type === 'quote' ? 'Valable jusqu\'au :' : data.type === 'invoice' ? 'Échéance :' : 'Date :'
   326→  const dueValue = data.type === 'quote'
   327→    ? formatDate(new Date(Date.now() + 30 * 24 * 60 * 60 * 1000))

   329→  page.drawText(dueLabel, { x: 320, y: boxY - 32, size: 9, font: fontBold, color: COLORS.gray })
   330→  page.drawText(dueValue, { x: 410, y: boxY - 32, size: 9, font: fontRegular, color: COLORS.dark })
   331→
   332→  // Status
   333→  const statusLabel = data.type === 'receipt' ? 'Payé le :' : 'Statut :'
   334→  const statusValue = data.type === 'receipt'
   335→    ? formatDate(data.paidAt || new Date())
   336→    : data.type === 'invoice' && data.paidAt
   337→      ? `Payé le ${formatDate(data.paidAt)}`
   338→      : 'En attente'
   339→  page.drawText(statusLabel, { x: 64, y: boxY - 48, size: 9, font: fontBold, color: COLORS.gray })
   340→  page.drawText(statusValue, { x: 160, y: boxY - 48, size: 9, font: fontBold, color: data.paidAt ? COLORS.primary : COLORS.amber })
   341→
   342→  return boxY - boxHeight - 20
   343→}

   345→function drawItemsTable(page: PDFPage, fontBold: PDFFont, fontRegular: PDFFont, data: DocumentData, startY: number): number {
   346→  const { width } = page.getSize()
   347→  const tableX = 50
   348→  const tableWidth = width - 100
   349→  let yPos = startY
   350→
   351→  // Table header
   352→  page.drawRectangle({
   353→    x: tableX,
   354→    y: yPos - 22,
   355→    width: tableWidth,
   356→    height: 22,
   357→    color: COLORS.primary,
   358→  })
   359→
   360→  page.drawText('Description', { x: tableX + 12, y: yPos - 15, size: 9, font: fontBold, color: COLORS.white })
   361→  page.drawText('Qté', { x: tableX + tableWidth - 230, y: yPos - 15, size: 9, font: fontBold, color: COLORS.white })
   362→  page.drawText('Prix unitaire', { x: tableX + tableWidth - 170, y: yPos - 15, size: 9, font: fontBold, color: COLORS.white })
   363→  page.drawText('Total', { x: tableX + tableWidth - 70, y: yPos - 15, size: 9, font: fontBold, color: COLORS.white })
   364→
   365→  yPos -= 22
   366→
   367→  // Items rows
   368→  let altRow = false
   369→  let subtotal = 0
   370→  for (const item of data.items) {
   371→    const rowHeight = 28
   372→    if (altRow) {
   373→      page.drawRectangle({
   374→        x: tableX,
   375→        y: yPos - rowHeight,
   376→        width: tableWidth,
   377→        height: rowHeight,
   378→        color: COLORS.veryLight,
   379→      })
   380→    }
   381→    altRow = !altRow
   382→
   383→    // Description (truncated if too long)
   384→    const desc = item.description.length > 65 ? item.description.slice(0, 62) + '...' : item.description
   385→    page.drawText(desc, { x: tableX + 12, y: yPos - 18, size: 9, font: fontRegular, color: COLORS.dark })
   386→
   387→    page.drawText(String(item.quantity), { x: tableX + tableWidth - 230, y: yPos - 18, size: 9, font: fontRegular, color: COLORS.dark })
   388→    page.drawText(formatMoney(item.unitPrice, data.currency), { x: tableX + tableWidth - 170, y: yPos - 18, size: 9, font: fontRegular, color: COLORS.dark })
   389→    page.drawText(formatMoney(item.total, data.currency), { x: tableX + tableWidth - 70, y: yPos - 18, size: 9, font: fontBold, color: COLORS.dark })
   390→
   391→    subtotal += item.total
   392→    yPos -= rowHeight
   393→  }
   394→
   395→  // Border around table
   396→  page.drawRectangle({
   397→    x: tableX,
   398→    y: yPos,
   399→    width: tableWidth,
   400→    height: startY - yPos,
   401→    borderColor: COLORS.border,
   402→    borderWidth: 1,
   403→  })
   404→
   405→  // Totals
   406→  const taxRate = data.taxRate || 0
   407→  const taxAmount = subtotal * (taxRate / 100)
   408→  const total = subtotal + taxAmount
   409→
   410→  yPos -= 20
   411→  const totalsX = tableX + tableWidth - 240
   412→
   413→  // Subtotal
   414→  page.drawText('Sous-total :', { x: totalsX, y: yPos, size: 10, font: fontRegular, color: COLORS.gray })
   415→  page.drawText(formatMoney(subtotal, data.currency), { x: tableX + tableWidth - 12 - 90, y: yPos, size: 10, font: fontRegular, color: COLORS.dark, align: TextAlignment.Right })
   416→
   417→  yPos -= 18
   418→  if (taxRate > 0) {
   419→    page.drawText(`TVA (${taxRate}%) :`, { x: totalsX, y: yPos, size: 10, font: fontRegular, color: COLORS.gray })
   420→    page.drawText(formatMoney(taxAmount, data.currency), { x: tableX + tableWidth - 12 - 90, y: yPos, size: 10, font: fontRegular, color: COLORS.dark, align: TextAlignment.Right })
   421→    yPos -= 18
   422→  }
   423→
   424→  // Total (highlighted)
   425→  page.drawRectangle({
   426→    x: totalsX - 12,
   427→    y: yPos - 6,
   428→    width: tableX + tableWidth - totalsX,
   429→    height: 26,
   430→    color: COLORS.primaryLight,
   431→  })
   432→  page.drawText('TOTAL :', { x: totalsX, y: yPos, size: 12, font: fontBold, color: COLORS.primaryDark })
   433→  page.drawText(formatMoney(total, data.currency), { x: tableX + tableWidth - 12 - 90, y: yPos, size: 12, font: fontBold, color: COLORS.primaryDark, align: TextAlignment.Right })
   434→
   435→  return yPos - 40
   436→}

   497→async function buildPdf(rawData: DocumentData): Promise<Uint8Array> {
   498→  // Sanitize user-provided data for PDF metadata (setTitle, setSubject, etc.)
   499→  const data: DocumentData = {
   500→    ...rawData,
   501→    recipientName: sanitizeText(rawData.recipientName),
   502→    recipientEmail: sanitizeText(rawData.recipientEmail),
   503→    recipientCompany: sanitizeText(rawData.recipientCompany),
   504→    recipientAddress: sanitizeText(rawData.recipientAddress),
   505→    recipientCountry: sanitizeText(rawData.recipientCountry),
   506→    subject: sanitizeText(rawData.subject),
   507→    notes: sanitizeText(rawData.notes),
   508→    agreementTerms: sanitizeText(rawData.agreementTerms),
   509→    items: rawData.items.map((item) => ({
   510→      ...item,
   511→      description: sanitizeText(item.description),
   512→    })),
   513→  }
   514→
   515→  const pdfDoc = await PDFDocument.create()
   516→  const fontRegular = await pdfDoc.embedFont(StandardFonts.Helvetica)
   517→  const fontBold = await pdfDoc.embedFont(StandardFonts.HelveticaBold)
   518→  const fontItalic = await pdfDoc.embedFont(StandardFonts.HelveticaOblique)
   519→
   520→  pdfDoc.setTitle(sanitizeText(`${TYPE_META[data.type].label} ${data.number} - HireNova`))
   521→  pdfDoc.setAuthor(sanitizeText('HireNova - E-Society 2050'))
   522→  pdfDoc.setSubject(data.subject)
   523→  pdfDoc.setCreator('HireNova Document Engine')
   524→  pdfDoc.setProducer('HireNova Auto-Generator')
   525→  pdfDoc.setCreationDate(new Date())
   526→
   527→  const page = pdfDoc.addPage([595.28, 841.89]) // A4
   528→
   529→  // CRITICAL: Override drawText to auto-sanitize ALL text for WinAnsi encoding.
   530→  // This prevents "WinAnsi cannot encode" errors with Unicode chars (em dash, narrow no-break space, accented chars, etc.)
   531→  // Also handles right-alignment manually since pdf-lib's drawText doesn't support align natively.
   532→  const originalDrawText = page.drawText.bind(page)
   533→  ;(page as any).drawText = (text: string, options: any) => {
   534→    const sanitized = sanitizeText(text)
   535→    if (options.align === TextAlignment.Right) {
   536→      const textWidth = options.font.widthOfTextAtSize(sanitized, options.size)
   537→      const { align, ...rest } = options
   538→      return originalDrawText(sanitized, { ...rest, x: options.x - textWidth })
   539→    }
   540→    const { align, ...rest } = options
   541→    return originalDrawText(sanitized, rest)
   542→  }
   543→
   544→  const meta = TYPE_META[data.type]
   545→  let yPos = drawHeader(page, fontBold, fontRegular, data, meta.label, meta.color)
   546→  yPos = drawMetaBox(page, fontBold, fontRegular, data, yPos)
   547→
   548→  // For agreement type, draw contract clauses instead of items table
   549→  if (data.type === 'agreement' && data.agreementTerms) {
   550→    yPos = drawAgreementClauses(page, fontBold, fontRegular, fontItalic, data, yPos)
   551→  } else {
   552→    yPos = drawItemsTable(page, fontBold, fontRegular, data, yPos)
   553→  }
   554→
   555→  // Payment terms / acceptance block
   556→  if (data.type === 'quote') {
   557→    yPos = drawAcceptanceBlock(page, fontBold, fontRegular, yPos)
   558→  } else if (data.type === 'invoice') {
   559→    yPos = drawPaymentTerms(page, fontBold, fontRegular, data, yPos)
   560→  }
   561→
   562→  drawFooter(page, fontBold, fontRegular, data)
   563→
   564→  const pdfBytes = await pdfDoc.save()
   565→  return pdfBytes
   566→}

   568→function drawAgreementClauses(
   569→  page: PDFPage,
   570→  fontBold: PDFFont,
   571→  fontRegular: PDFFont,
   572→  fontItalic: PDFFont,
   573→  data: DocumentData,
   574→  startY: number
   575→): number {
   576→  let yPos = startY
   577→
   578→  page.drawText('CLAUSES DU CONTRAT', { x: 50, y: yPos, size: 11, font: fontBold, color: COLORS.primaryDark })
   579→  yPos -= 20
   580→
   581→  const terms = data.agreementTerms || ''
   582→  const clauses = terms.split('\n\n').filter(Boolean)
   583→
   584→  for (let i = 0; i < clauses.length; i++) {
   585→    const clause = clauses[i]
   586→    const title = `Article ${i + 1}`
   587→    page.drawText(title, { x: 50, y: yPos, size: 10, font: fontBold, color: COLORS.dark })
   588→    yPos -= 14
   589→
   590→    // Wrap text
   591→    const maxWidth = 495
   592→    const words = clause.split(' ')
   593→    let line = ''
   594→    const lines: string[] = []
   595→    for (const word of words) {
   596→      const testLine = line ? line + ' ' + word : word
   597→      try {
   598→        const testWidth = fontRegular.widthOfTextAtSize(testLine, 9)
   599→        if (testWidth > maxWidth) {
   600→          if (line) lines.push(line)
   601→          line = word
   602→        } else {
   603→          line = testLine
   604→        }
   605→      } catch {
   606→        line = line ? line + ' ' + word : word
   607→      }
   608→    }
   609→    if (line) lines.push(line)
   610→
   611→    for (const ln of lines.slice(0, 6)) {
   612→      page.drawText(ln.slice(0, 100), { x: 60, y: yPos, size: 9, font: fontRegular, color: COLORS.gray })
   613→      yPos -= 12
   614→    }
   615→    yPos -= 8
   616→    if (yPos < 180) break
   617→  }
   618→
   619→  // Signatures
   620→  yPos = Math.max(yPos, 200)
   621→  page.drawText('Signatures', { x: 50, y: yPos, size: 11, font: fontBold, color: COLORS.dark })
   622→  yPos -= 20
   623→
   624→  page.drawLine({ start: { x: 50, y: yPos }, end: { x: 230, y: yPos }, thickness: 1, color: COLORS.gray })
   625→  page.drawLine({ start: { x: 320, y: yPos }, end: { x: 500, y: yPos }, thickness: 1, color: COLORS.gray })
   626→
   627→  page.drawText('Pour HireNova', { x: 50, y: yPos - 14, size: 9, font: fontBold, color: COLORS.dark })
   628→  page.drawText(data.recipientName, { x: 320, y: yPos - 14, size: 9, font: fontBold, color: COLORS.dark })
   629→
   630→  page.drawText('Date : _______________', { x: 50, y: yPos - 30, size: 9, font: fontRegular, color: COLORS.gray })
   631→  page.drawText('Date : _______________', { x: 320, y: yPos - 30, size: 9, font: fontRegular, color: COLORS.gray })
   632→
   633→  return yPos - 50
   634→}

   679→/**
   680→ * Generate a document PDF and persist it in the database.
   681→ * Returns the generated document with PDF base64.
   682→ */
   683→export async function generateDocument(data: DocumentData): Promise<GeneratedDocument> {
   684→  const pdfBytes = await buildPdf(data)
   685→  const pdfBase64 = Buffer.from(pdfBytes).toString('base64')
   686→
   687→  // Calculate totals
   688→  const subtotal = data.items.reduce((sum, item) => sum + item.total, 0)
   689→  const taxRate = data.taxRate || 0
   690→  const taxAmount = subtotal * (taxRate / 100)
   691→  const total = subtotal + taxAmount
   692→
   693→  // Persist to DB
   694→  const doc = await db.document.create({
   695→    data: {
   696→      type: data.type,
   697→      number: data.number,
   698→      recipientName: data.recipientName,
   699→      recipientEmail: data.recipientEmail,
   700→      recipientCompany: data.recipientCompany || null,
   701→      recipientAddress: data.recipientAddress || null,
   702→      recipientCountry: data.recipientCountry || null,
   703→      issuerName: data.issuerName || 'HireNova',
   704→      issuerLegal: data.issuerLegal || 'E-Society 2050',
   705→      issuerAddress: data.issuerAddress || 'Casablanca, Maroc',
   706→      issuerEmail: data.issuerEmail || 'hello@hirenova.com',
   707→      issuerVAT: data.issuerVAT,
   708→      subject: data.subject,
   709→      items: JSON.stringify(data.items),
   710→      currency: data.currency,
   711→      subtotal,
   712→      taxRate,
   713→      taxAmount,
   714→      total,
   715→      status: 'draft',
   716→      issueDate: data.issueDate || new Date(),
   717→      dueDate: data.dueDate || null,
   718→      paidAt: data.paidAt || null,
   719→      userId: data.userId || null,
   720→      inquiryId: data.inquiryId || null,
   721→      pdfBase64,
   722→      notes: data.notes || null,
   723→    },
   724→  })
   725→
   726→  return {
   727→    id: doc.id,
   728→    number: doc.number,
   729→    type: doc.type as DocumentType,
   730→    pdfBase64,
   731→    total,
   732→  }
   733→}
