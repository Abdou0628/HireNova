    92→
    93→  const [isSubmitting, setIsSubmitting] = useState(false)

    97→  const [isImportingCv, setIsImportingCv] = useState(false)
    98→  const fileInputRef = useRef<HTMLInputElement>(null)
    99→
   100→  async function handleImportCv(e: React.ChangeEvent<HTMLInputElement>) {
   101→    const file = e.target.files?.[0]
   102→    if (!file) return
   103→    setIsImportingCv(true)
   104→    try {
   105→      const formData = new FormData()
   106→      formData.append('file', file)
   107→      formData.append('language', language)
   108→      const res = await fetch('/api/import-cv', {
   109→        method: 'POST',
   110→        body: formData,
   111→      })
   112→      const data = await res.json()
   113→      if (!res.ok) throw new Error(data.error || 'Erreur lors de l\'import')
   114→      // Auto-fill form fields from parsed CV
   115→      if (data.fullName) updateFormData({ fullName: data.fullName })
   116→      if (data.email) updateFormData({ email: data.email })
   117→      if (data.phone) updateFormData({ phone: data.phone })
   118→      if (data.location) updateFormData({ location: data.location })
   119→      if (data.targetJob) updateFormData({ targetJob: data.targetJob })
   120→      if (data.industry) updateFormData({ industry: data.industry })
   121→      if (data.experience) updateFormData({ experience: data.experience })
   122→      if (data.education) updateFormData({ education: data.education })
   123→      if (data.skills) updateFormData({ skills: data.skills })
   124→      if (data.languages) updateFormData({ languages: data.languages })
   125→      if (data.summary) updateFormData({ summary: data.summary })
   126→      if (data.linkedin) updateFormData({ linkedin: data.linkedin })
   127→      if (data.website) updateFormData({ website: data.website })
   128→      toast.success(t(language, 'importCvSuccess'))
   129→    } catch (err) {
   130→      const message = err instanceof Error ? err.message : 'Erreur inconnue'
   131→      toast.error(message)
   132→    } finally {
   133→      setIsImportingCv(false)
   134→      if (fileInputRef.current) fileInputRef.current.value = ''
   135→    }
   136→  }
   137→
   138→  function handlePhoneCountryChange(code: string) {
   139→    setPhoneCountryCode(code)
   140→    const country = countries.find((c) => c.code === code)
   141→    if (country) {
   142→      // Remove any existing dial code from phone, prepend new one
   143→      const currentNumber = formData.phone.replace(/^[+\d\s\-()]+/, '').trim()
   144→      updateFormData({ phone: currentNumber ? `${country.dial} ${currentNumber}` : country.dial })
   145→    }
   146→  }
   147→
   148→  function handlePhoneNumberChange(number: string) {
   149→    const country = countries.find((c) => c.code === phoneCountryCode)
   150→    const dial = country?.dial || ''
   151→    const cleaned = number.replace(/[^\d\s]/g, '').trim()
   152→    updateFormData({ phone: dial ? `${dial} ${cleaned}` : cleaned })
   153→  }
   154→
   155→  function getPhoneNumberOnly(): string {
   156→    const country = countries.find((c) => c.code === phoneCountryCode)
   157→    const dial = country?.dial || ''
   158→    return formData.phone.startsWith(dial) ? formData.phone.slice(dial.length).trim() : formData.phone.replace(/^[+\d\s\-()]+/, '').trim()
   159→  }
   160→
   161→  const slideVariants = {
   162→    enter: (direction: number) => ({ x: direction > 0 ? 300 : -300, opacity: 0 }),
   163→    center: { x: 0, opacity: 1 },
   164→    exit: (direction: number) => ({ x: direction < 0 ? 300 : -300, opacity: 0 }),
   165→  }
   166→
   167→  function validateStep(step: number): boolean {
   168→    if (step === 0) {
   169→      if (!formData.fullName.trim() || !formData.email.trim()) {
   170→        toast.error(t(language, 'errorFillFields'))
   171→        return false
   172→      }
   173→    }
   174→    if (step === 1) {
   175→      if (!formData.targetJob.trim()) {
   176→        toast.error(t(language, 'errorFillFields'))
   177→        return false
   178→      }
   179→    }
   180→    return true
   181→  }
   182→
   183→  function handleNext() {
   184→    if (!validateStep(formStep)) return
   185→    if (formStep < 3) setFormStep(formStep + 1)
   186→  }
   187→
   188→  function handlePrevious() {
   189→    if (formStep > 0) setFormStep(formStep - 1)
   190→  }
   191→
   192→  async function handleGenerate() {

   195→    setStep('generating')

   197→    setError(null)

   199→    setCLError(null)
   200→

   202→

   204→      // Generate CV
   205→      const res = await fetch('/api/generate-cv', {
   206→        method: 'POST',
   207→        headers: { 'Content-Type': 'application/json' },
   208→        body: JSON.stringify({
   209→          ...formData,
   210→          language,
   211→          persona: selectedPersona || undefined,

   213→          internshipRequest: applicationType === 'internship' || undefined,
   214→        }),
   215→      })
   216→
   217→      const data = await res.json()

   219→        if (data.code === 'LIMIT_REACHED') {
   220→          setStep('landing')

   224→        throw new Error(data.error || 'Erreur lors de la génération')
   225→      }
   226→
   227→      const generatedCV = data.cv
   228→      setGeneratedCV(generatedCV)
   229→
   230→      // If company name provided, auto-generate cover letter simultaneously
   231→      if (hasCompany) {

   233→          // Auto-sync personal info to CL form data
   234→          updateCLFormData({
   235→            fullName: formData.fullName,
   236→            email: formData.email,
   237→            phone: formData.phone,
   238→            address: formData.address,
   239→            location: formData.location,
   240→            companyName: formData.companyName,
   241→            hiringManager: formData.hiringManager,
   242→            jobTitle: formData.targetJob,
   243→            tone: formData.clTone,
   244→          })
   245→
   246→          const clRes = await fetch('/api/generate-cover-letter', {
   247→            method: 'POST',
   248→            headers: { 'Content-Type': 'application/json' },
   249→            body: JSON.stringify({
   250→              fullName: formData.fullName,
   251→              email: formData.email,
   252→              phone: formData.phone,
   253→              address: formData.address,
   254→              location: formData.location,
   255→              companyName: formData.companyName,
   256→              hiringManager: formData.hiringManager,
   257→              jobTitle: formData.targetJob,
   258→              tone: formData.clTone,
   259→              language,
   260→              generatedCVSummary: generatedCV.summary,
   261→              generatedCVExperience: generatedCV.experience,
   262→              generatedCVEducation: generatedCV.education,
   263→              generatedCVSkills: generatedCV.skills,
   264→              generatedCVLanguages: generatedCV.languages,
   265→            }),
   266→          })
   267→
   268→          const clData = await clRes.json()
   269→          if (clRes.ok && clData.letter) {
   270→            setGeneratedCL(clData.letter)
   271→          }
   272→        } catch {
   273→          // CL generation failed silently - CV is still valid
   274→          setIsCLGenerating(false)
   275→        }
   276→      }
   277→
   278→      setTimeout(() => {
   279→        setStep('preview')
   280→        setIsGenerating(false)
   281→        setIsCLGenerating(false)

   283→    } catch (err) {
   284→      setError(err instanceof Error ? err.message : 'Erreur inconnue')
   285→      setIsGenerating(false)
   286→      setIsCLGenerating(false)
   287→      setStep('form')
   288→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue')
   289→    } finally {
   290→      setIsSubmitting(false)
   291→    }
   292→  }
   293→
   294→  const isLastStep = formStep === 3
   295→
   296→  return (
   297→    <div className="min-h-screen flex flex-col bg-background">
   298→      {/* Header */}
   299→      <header className="w-full px-4 sm:px-6 lg:px-8 py-4 border-b bg-white/80 backdrop-blur-sm sticky top-0 z-50">
   300→        <div className="max-w-3xl mx-auto flex items-center justify-between">
   301→          <button
   302→            onClick={() => { setStep('landing'); setSelectedPersona(null) }}
   303→            className="flex items-center gap-2 text-muted-foreground hover:text-foreground transition-colors"
   304→          >
   305→            <div className="flex flex-col items-center">
   306→              <Image src="/hirenova-logo.png" alt="HireNova" width={32} height={32} className="rounded-lg" />
   307→              <span className="text-[9px] font-semibold text-emerald-600 tracking-wide">POWERED BY IA</span>
   308→            </div>
   309→            <span className="font-semibold text-foreground hidden sm:inline">{t(language, 'siteTitle')}</span>
   310→          </button>
   311→          {selectedPersona && (
   312→            <span className="flex items-center gap-1.5 text-xs font-medium bg-emerald-50 text-emerald-700 px-2.5 py-1 rounded-full border border-emerald-200">
   313→              <span>{personaEmoji[selectedPersona]}</span>
   314→              <span>{t(language, personaNameKey[selectedPersona])}</span>
   315→            </span>
   316→          )}
   317→          <div className="flex items-center gap-1 bg-muted rounded-lg p-1">
   318→            {languages.map((lang) => (
   319→              <button
   320→                key={lang.id}
   321→                onClick={() => setLanguage(lang.id)}
   322→                className={`px-3 py-1.5 rounded-md text-sm font-medium transition-all ${
   323→                  language === lang.id
   324→                    ? 'bg-white shadow-sm text-foreground'
   325→                    : 'text-muted-foreground hover:text-foreground'
   326→                }`}
   327→              >
   328→                <span className="mr-1">{lang.flag}</span>
   329→                <span className="hidden sm:inline">{lang.id.toUpperCase()}</span>
   330→              </button>
   331→            ))}
   332→          </div>
   333→        </div>
   334→      </header>
   335→
   336→      <main className="flex-1 py-8 px-4 sm:px-6">
   337→        <div className="max-w-2xl mx-auto">
   338→          {/* Progress */}
   339→          <div className="mb-8">
   340→            <div className="flex items-center justify-between mb-4">
   341→              {stepTitles.map((titleKey, i) => {
   342→                const Icon = stepIcons[i]
   343→                const isActive = i === formStep
   344→                const isCompleted = i < formStep
   345→                return (
   346→                  <button
   347→                    key={titleKey}
   348→                    onClick={() => {
   349→                      if (isCompleted || i === formStep) setFormStep(i)
   350→                    }}
   351→                    className={`flex flex-col items-center gap-1.5 flex-1 cursor-pointer group ${
   352→                      !isCompleted && i !== formStep ? 'opacity-40' : ''
   353→                    }`}
   354→                  >
   355→                    <div
   356→                      className={`w-10 h-10 rounded-full flex items-center justify-center transition-all ${
   357→                        isActive
   358→                          ? 'bg-emerald-600 text-white shadow-lg shadow-emerald-600/25'
   359→                          : isCompleted
   360→                            ? 'bg-emerald-100 text-emerald-700'
   361→                            : 'bg-muted text-muted-foreground'
   362→                      }`}
   363→                    >
   364→                      <Icon className="w-5 h-5" />
   365→                    </div>
   366→                    <span
   367→                      className={`text-xs font-medium text-center hidden sm:block ${
   368→                        isActive ? 'text-emerald-600' : 'text-muted-foreground'
   369→                      }`}
   370→                    >
   371→                      {t(language, titleKey)}
   372→                    </span>
   373→                  </button>
   374→                )

   376→            </div>
   377→            {/* Progress bar */}
   378→            <div className="h-1.5 bg-muted rounded-full overflow-hidden">
   379→              <motion.div
   380→                className="h-full bg-emerald-500 rounded-full"
   381→                animate={{ width: `${((formStep + 1) / 4) * 100}%` }}
   382→                transition={{ duration: 0.4, ease: 'easeOut' }}
   383→              />
   384→            </div>
   385→          </div>
   386→
   387→          {/* Form Content */}
   388→          <div className="bg-white rounded-2xl border shadow-sm p-6 sm:p-8 min-h-[400px] relative overflow-hidden">
   389→            <AnimatePresence mode="wait" custom={1}>
   390→              <motion.div
   391→                key={formStep}
   392→                custom={1}
   393→                variants={slideVariants}
   394→                initial="enter"
   395→                animate="center"
   396→                exit="exit"
   397→                transition={{ duration: 0.3, ease: 'easeInOut' }}
   398→              >
   399→                <h2 className="text-xl font-bold text-foreground mb-6 flex items-center gap-2">
   400→                  {(() => {
   401→                    const Icon = stepIcons[formStep]
   402→                    return <Icon className="w-5 h-5 text-emerald-600" />
   403→                  })()}
   404→                  {t(language, stepTitles[formStep])}
   405→                </h2>
   406→
   407→                {/* Step 1: Personal Info */}
   408→                {formStep === 0 && (
   409→                  <div className="space-y-4">
   410→                    {/* Import Existing CV Banner */}
   411→                    <div className="rounded-xl border-2 border-dashed border-emerald-300 bg-gradient-to-r from-emerald-50/60 to-teal-50/40 p-5">
   412→                      <div className="flex items-start gap-3">
   413→                        <div className="w-10 h-10 rounded-xl bg-emerald-100 flex items-center justify-center shrink-0">
   414→                          <Upload className="w-5 h-5 text-emerald-600" />
   415→                        </div>
   416→                        <div className="flex-1 min-w-0">
   417→                          <h3 className="text-sm font-semibold text-emerald-800 mb-1">{t(language, 'importCvTitle')}</h3>
   418→                          <p className="text-xs text-emerald-700/80 mb-3 leading-relaxed">{t(language, 'importCvDesc')}</p>
   419→                          <div className="flex items-center gap-3">
   420→                            <Button
   421→                              type="button"
   422→                              variant="outline"
   423→                              size="sm"
   424→                              onClick={() => fileInputRef.current?.click()}
   425→                              disabled={isImportingCv}
   426→                              className="bg-emerald-600 hover:bg-emerald-700 text-white border-emerald-600 gap-2 cursor-pointer text-xs font-medium"
   427→                            >
   428→                              {isImportingCv ? (
   429→                                <Loader2 className="w-3.5 h-3.5 animate-spin" />
   430→                              ) : (
   431→                                <FileUp className="w-3.5 h-3.5" />
   432→                              )}
   433→                              {isImportingCv ? t(language, 'importCvParsing') : t(language, 'importCvBtn')}
   434→                            </Button>
   435→                            <span className="text-[10px] text-muted-foreground">PDF · DOCX · TXT</span>
   436→                          </div>
   437→                        </div>
   438→                      </div>
   439→                      <input
   440→                        ref={fileInputRef}
   441→                        type="file"
   442→                        accept=".pdf,.doc,.docx,.txt"
   443→                        className="hidden"
   444→                        onChange={handleImportCv}
   445→                      />
   446→                    </div>
   447→
   448→                    {/* Photo Upload */}
   449→                    <div>
   450→                      <Label>{t(language, 'photo')}</Label>
   451→                      <div className="mt-1.5">
   452→                        {formData.photo ? (
   453→                          <div className="relative inline-block">
   454→                            <img
   455→                              src={formData.photo}
   456→                              alt="Photo"
   457→                              className="w-24 h-24 rounded-full object-cover border-2 border-emerald-200"
   458→                            />
   459→                            <button
   460→                              type="button"
   461→                              onClick={() => updateFormData({ photo: '' })}
   462→                              className="absolute -top-1 -right-1 w-6 h-6 rounded-full bg-red-500 text-white flex items-center justify-center hover:bg-red-600 transition-colors cursor-pointer shadow-sm"
   463→                            >
   464→                              <X className="w-3 h-3" />
   465→                            </button>
   466→                          </div>
   467→                        ) : (
   468→                          <label className="flex flex-col items-center justify-center w-full h-32 border-2 border-dashed border-muted-foreground/25 rounded-xl cursor-pointer hover:border-emerald-400 hover:bg-emerald-50/50 transition-all group">
   469→                            <input
   470→                              type="file"
   471→                              accept="image/*"
   472→                              className="hidden"
   473→                              onChange={(e) => {
   474→                                const file = e.target.files?.[0]
   475→                                if (!file) return
   476→                                if (file.size > 5 * 1024 * 1024) {
   477→                                  toast.error(language === 'fr' ? 'Image trop volumineuse (max 5 Mo)' : language === 'en' ? 'Image too large (max 5 MB)' : 'الصورة كبيرة جداً (الحد الأقصى 5 ميغابايت)')
   478→                                  return
   479→                                }
   480→                                const reader = new FileReader()
   481→                                reader.onload = () => {
   482→                                  updateFormData({ photo: reader.result as string })
   483→                                }
   484→                                reader.readAsDataURL(file)
   485→                              }}
   486→                            />
   487→                            <Camera className="w-8 h-8 text-muted-foreground/40 group-hover:text-emerald-500 transition-colors" />
   488→                            <span className="text-xs text-muted-foreground/60 mt-1.5 group-hover:text-emerald-600 transition-colors">
   489→                              {t(language, 'photoPlaceholder')}
   490→                            </span>
   491→                          </label>
   492→                        )}
   493→                      </div>
   494→                      <div className="flex items-start gap-1.5 mt-2">
   495→                        <Info className="w-3.5 h-3.5 text-amber-500 shrink-0 mt-0.5" />
   496→                        <p className="text-xs text-muted-foreground">{t(language, 'photoNote')}</p>
   497→                      </div>
   498→                    </div>
