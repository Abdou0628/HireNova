    53→]
    54→
    55→export default function CVForm() {
    56→  const {
    57→    formStep,
    58→    setFormStep,
    59→    language,
    60→    setLanguage,
    61→    template,
    62→    setTemplate,
    63→    formData,
    64→    updateFormData,
    65→    setStep,
    66→    setIsGenerating,
    67→    setGeneratedCV,
    68→    setError,
    69→    updateCLFormData,
    70→    setGeneratedCL,
    71→    setCLGenerating,
    72→    setCLError,
    73→  } = useCVStore()
    74→
    75→  const [isSubmitting, setIsSubmitting] = useState(false)
    76→  const [phoneCountryCode, setPhoneCountryCode] = useState<string>('')
    77→
    78→  function handlePhoneCountryChange(code: string) {
    79→    setPhoneCountryCode(code)
    80→    const country = countries.find((c) => c.code === code)
    81→    if (country) {
    82→      // Remove any existing dial code from phone, prepend new one
    83→      const currentNumber = formData.phone.replace(/^[+\d\s\-()]+/, '').trim()
    84→      updateFormData({ phone: currentNumber ? `${country.dial} ${currentNumber}` : country.dial })
    85→    }
    86→  }
    87→
    88→  function handlePhoneNumberChange(number: string) {
    89→    const country = countries.find((c) => c.code === phoneCountryCode)
    90→    const dial = country?.dial || ''
    91→    const cleaned = number.replace(/[^\d\s]/g, '').trim()
    92→    updateFormData({ phone: dial ? `${dial} ${cleaned}` : cleaned })
    93→  }
    94→
    95→  function getPhoneNumberOnly(): string {
    96→    const country = countries.find((c) => c.code === phoneCountryCode)
    97→    const dial = country?.dial || ''
    98→    return formData.phone.startsWith(dial) ? formData.phone.slice(dial.length).trim() : formData.phone.replace(/^[+\d\s\-()]+/, '').trim()
    99→  }
   100→
   101→  const slideVariants = {
   102→    enter: (direction: number) => ({ x: direction > 0 ? 300 : -300, opacity: 0 }),
   103→    center: { x: 0, opacity: 1 },
   104→    exit: (direction: number) => ({ x: direction < 0 ? 300 : -300, opacity: 0 }),
   105→  }
   106→
   107→  function validateStep(step: number): boolean {
   108→    if (step === 0) {
   109→      if (!formData.fullName.trim() || !formData.email.trim()) {
   110→        toast.error(t(language, 'errorFillFields'))
   111→        return false
   112→      }
   113→    }
   114→    if (step === 1) {
   115→      if (!formData.targetJob.trim()) {
   116→        toast.error(t(language, 'errorFillFields'))
   117→        return false
   118→      }
   119→    }
   120→    return true
   121→  }
   122→
   123→  function handleNext() {
   124→    if (!validateStep(formStep)) return
   125→    if (formStep < 3) setFormStep(formStep + 1)
   126→  }
   127→
   128→  function handlePrevious() {
   129→    if (formStep > 0) setFormStep(formStep - 1)
   130→  }
   131→
   132→  async function handleGenerate() {
   133→    if (!validateStep(formStep)) return
   134→    setIsSubmitting(true)
   135→    setStep('generating')

   146→        method: 'POST',
   147→        headers: { 'Content-Type': 'application/json' },
   148→        body: JSON.stringify({ ...formData, language }),
   149→      })
   150→
   151→      const data = await res.json()

   153→        throw new Error(data.error || 'Erreur lors de la génération')
   154→      }
   155→
   156→      const generatedCV = data.cv
   157→      setGeneratedCV(generatedCV)
   158→
   159→      // If company name provided, auto-generate cover letter simultaneously

   161→        try {
   162→          // Auto-sync personal info to CL form data
   163→          updateCLFormData({
   164→            fullName: formData.fullName,
   165→            email: formData.email,
   166→            phone: formData.phone,
   167→            address: formData.address,
   168→            location: formData.location,
   169→            companyName: formData.companyName,
   170→            hiringManager: formData.hiringManager,
   171→            jobTitle: formData.targetJob,
   172→            tone: formData.clTone,
   173→          })
   174→
   175→          const clRes = await fetch('/api/generate-cover-letter', {
   176→            method: 'POST',
   177→            headers: { 'Content-Type': 'application/json' },
   178→            body: JSON.stringify({
   179→              fullName: formData.fullName,
   180→              email: formData.email,
   181→              phone: formData.phone,
   182→              address: formData.address,
   183→              location: formData.location,
   184→              companyName: formData.companyName,
   185→              hiringManager: formData.hiringManager,
   186→              jobTitle: formData.targetJob,
   187→              tone: formData.clTone,
   188→              language,
   189→              generatedCVSummary: generatedCV.summary,
   190→              generatedCVExperience: generatedCV.experience,
   191→              generatedCVEducation: generatedCV.education,
   192→              generatedCVSkills: generatedCV.skills,
   193→              generatedCVLanguages: generatedCV.languages,
   194→            }),
   195→          })
   196→
   197→          const clData = await clRes.json()
   198→          if (clRes.ok && clData.letter) {
   199→            setGeneratedCL(clData.letter)
   200→          }

   202→          // CL generation failed silently - CV is still valid
   203→          setCLGenerating(false)
   204→        }
   205→      }
   206→
   207→      setTimeout(() => {
   208→        setStep('preview')
   209→        setIsGenerating(false)
   210→        setCLGenerating(false)

   213→      setError(err instanceof Error ? err.message : 'Erreur inconnue')
   214→      setIsGenerating(false)
   215→      setCLGenerating(false)
   216→      setStep('form')

   219→      setIsSubmitting(false)
   220→    }
   221→  }

   229→        <div className="max-w-3xl mx-auto flex items-center justify-between">
   230→          <button
   231→            onClick={() => setStep('landing')}
   232→            className="flex items-center gap-2 text-muted-foreground hover:text-foreground transition-colors"
   233→          >
   234→            <div className="w-8 h-8 rounded-lg bg-emerald-600 flex items-center justify-center">
   235→              <FileText className="w-4 h-4 text-white" />

   278→                      className={`w-10 h-10 rounded-full flex items-center justify-center transition-all ${
   279→                        isActive
   280→                          ? 'bg-emerald-600 text-white shadow-lg shadow-emerald-600/25'
   281→                          : isCompleted
   282→                            ? 'bg-emerald-100 text-emerald-700'
   283→                            : 'bg-muted text-muted-foreground'
   284→                      }`}
   285→                    >
   286→                      <Icon className="w-5 h-5" />
   287→                    </div>
   288→                    <span
   289→                      className={`text-xs font-medium text-center hidden sm:block ${
   290→                        isActive ? 'text-emerald-600' : 'text-muted-foreground'
   291→                      }`}
   292→                    >
   293→                      {t(language, titleKey)}
   294→                    </span>
   295→                  </button>
   296→                )
   297→              })}
   298→            </div>
   299→            {/* Progress bar */}
   300→            <div className="h-1.5 bg-muted rounded-full overflow-hidden">
   301→              <motion.div
   302→                className="h-full bg-emerald-500 rounded-full"
   303→                animate={{ width: `${((formStep + 1) / 4) * 100}%` }}
   304→                transition={{ duration: 0.4, ease: 'easeOut' }}
   305→              />
   306→            </div>
   307→          </div>
   308→
   309→          {/* Form Content */}
   310→          <div className="bg-white rounded-2xl border shadow-sm p-6 sm:p-8 min-h-[400px] relative overflow-hidden">
   311→            <AnimatePresence mode="wait" custom={1}>
   312→              <motion.div
   313→                key={formStep}
   314→                custom={1}
   315→                variants={slideVariants}
   316→                initial="enter"
   317→                animate="center"
   318→                exit="exit"
   319→                transition={{ duration: 0.3, ease: 'easeInOut' }}
   320→              >
   321→                <h2 className="text-xl font-bold text-foreground mb-6 flex items-center gap-2">
   322→                  {(() => {
   323→                    const Icon = stepIcons[formStep]
   324→                    return <Icon className="w-5 h-5 text-emerald-600" />
   325→                  })()}
   326→                  {t(language, stepTitles[formStep])}
   327→                </h2>
   328→
   329→                {/* Step 1: Personal Info */}
   330→                {formStep === 0 && (
   331→                  <div className="space-y-4">
   332→                    {/* Photo Upload */}
   333→                    <div>
   334→                      <Label>{t(language, 'photo')}</Label>
   335→                      <div className="mt-1.5">
   336→                        {formData.photo ? (
   337→                          <div className="relative inline-block">
   338→                            <img
   339→                              src={formData.photo}
   340→                              alt="Photo"
   341→                              className="w-24 h-24 rounded-full object-cover border-2 border-emerald-200"
   342→                            />
   343→                            <button
   344→                              type="button"
   345→                              onClick={() => updateFormData({ photo: '' })}
   346→                              className="absolute -top-1 -right-1 w-6 h-6 rounded-full bg-red-500 text-white flex items-center justify-center hover:bg-red-600 transition-colors cursor-pointer shadow-sm"
   347→                            >
   348→                              <X className="w-3 h-3" />
   349→                            </button>
   350→                          </div>
   351→                        ) : (
   352→                          <label className="flex flex-col items-center justify-center w-full h-32 border-2 border-dashed border-muted-foreground/25 rounded-xl cursor-pointer hover:border-emerald-400 hover:bg-emerald-50/50 transition-all group">
   353→                            <input
   354→                              type="file"
   355→                              accept="image/*"
   356→                              className="hidden"
   357→                              onChange={(e) => {
   358→                                const file = e.target.files?.[0]
   359→                                if (!file) return
   360→                                if (file.size > 5 * 1024 * 1024) {
   361→                                  toast.error(language === 'fr' ? 'Image trop volumineuse (max 5 Mo)' : language === 'en' ? 'Image too large (max 5 MB)' : 'الصورة كبيرة جداً (الحد الأقصى 5 ميغابايت)')
   362→                                  return
   363→                                }
   364→                                const reader = new FileReader()
   365→                                reader.onload = () => {
   366→                                  updateFormData({ photo: reader.result as string })
   367→                                }
   368→                                reader.readAsDataURL(file)
   369→                              }}
   370→                            />
   371→                            <Camera className="w-8 h-8 text-muted-foreground/40 group-hover:text-emerald-500 transition-colors" />
   372→                            <span className="text-xs text-muted-foreground/60 mt-1.5 group-hover:text-emerald-600 transition-colors">
   373→                              {t(language, 'photoPlaceholder')}
   374→                            </span>
   375→                          </label>
   376→                        )}
   377→                      </div>
   378→                      <div className="flex items-start gap-1.5 mt-2">
   379→                        <Info className="w-3.5 h-3.5 text-amber-500 shrink-0 mt-0.5" />
   380→                        <p className="text-xs text-muted-foreground">{t(language, 'photoNote')}</p>
   381→                      </div>
   382→                    </div>
   383→                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   384→                      <div className="sm:col-span-2">
   385→                        <Label htmlFor="fullName">{t(language, 'fullName')} *</Label>
   386→                        <Input
   387→                          id="fullName"
   388→                          value={formData.fullName}
   389→                          onChange={(e) => updateFormData({ fullName: e.target.value })}
   390→                          placeholder={language === 'fr' ? 'Jean Dupont' : language === 'en' ? 'John Smith' : 'محمد أحمد'}
   391→                          className="mt-1.5"
   392→                        />
   393→                      </div>
   394→                      <div>
   395→                        <Label htmlFor="email">{t(language, 'email')} *</Label>
   396→                        <Input
   397→                          id="email"
   398→                          type="email"
   399→                          value={formData.email}
   400→                          onChange={(e) => updateFormData({ email: e.target.value })}
   401→                          placeholder="jean@exemple.com"
   402→                          className="mt-1.5"
   403→                        />
   404→                      </div>
   405→                      <div className="sm:col-span-2">
   406→                        <Label>{t(language, 'phone')}</Label>

   408→                          <select
   409→                            value={phoneCountryCode}

   411→                            className="h-9 w-[180px] shrink-0 rounded-md border border-input bg-background px-3 text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500 focus:ring-offset-1 appearance-none cursor-pointer"
   412→                            aria-label={t(language, 'phoneCountry')}
   413→                          >
   414→                            <option value="">{t(language, 'phoneCountryPlaceholder')}</option>
   415→                            {countries.map((c) => (
   416→                              <option key={c.code} value={c.code}>
   417→                                {c.flag} {c.dial} {c.name}
   418→                              </option>
   419→                            ))}
   420→                          </select>
   421→                          <Input
   422→                            type="tel"
   423→                            value={getPhoneNumberOnly()}
   424→                            onChange={(e) => handlePhoneNumberChange(e.target.value)}
   425→                            placeholder={language === 'fr' ? '600 000 000' : language === 'en' ? '600 000 000' : language === 'ar' ? '600 000 000' : '600 000 000'}
   426→                            className="flex-1"
   427→                          />
   428→                        </div>
   429→                      </div>
   430→                      <div>
   431→                        <Label htmlFor="address">{t(language, 'address')}</Label>
   432→                        <Input
   433→                          id="address"
   434→                          value={formData.address}
   435→                          onChange={(e) => updateFormData({ address: e.target.value })}
   436→                          placeholder={language === 'fr' ? '123 Rue Mohammed V' : language === 'en' ? '123 Main Street' : language === 'ar' ? 'شارع محمد الخامس 123' : 'Calle Ejemplo 123'}
   437→                          className="mt-1.5"
   438→                        />
   439→                      </div>
   440→                      <div>
   441→                        <Label htmlFor="location">{t(language, 'location')}</Label>
   442→                        <Input
   443→                          id="location"
   444→                          value={formData.location}
   445→                          onChange={(e) => updateFormData({ location: e.target.value })}
   446→                          placeholder={language === 'fr' ? 'Casablanca, Maroc' : language === 'en' ? 'London, UK' : language === 'ar' ? 'الدار البيضاء، المغرب' : 'Madrid, España'}
   447→                          className="mt-1.5"

   469→                      </div>
   470→                    </div>
   471→
   472→                    {/* Birth Information */}
   473→                    <div className="mt-6 pt-5 border-t border-stone-100">
   474→                      <Label className="flex items-center gap-2 text-sm font-semibold text-stone-700 mb-3">
   475→                        <CalendarDays className="w-4 h-4 text-emerald-600" />
   476→                        {t(language, 'personalInfo')}
   477→                      </Label>

   479→                        <div>
   480→                          <Label htmlFor="dateOfBirth">{t(language, 'dateOfBirth')}</Label>

   486→                            className="mt-1.5"
   487→                          />

   490→                          <Label htmlFor="birthPlace">{t(language, 'birthPlace')}</Label>

   493→                            value={formData.birthPlace}

   514→                {/* Step 2: Career Goals & Cover Letter */}
