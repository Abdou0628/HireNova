   115→/**
   116→ * Generate the next sequential document number: FAC-2026-001
   117→ */
   118→export async function nextDocumentNumber(type: DocumentType, date = new Date()): Promise<string> {
   119→  const year = date.getFullYear()
   120→  const prefix = TYPE_PREFIXES[type]
   121→
   122→  // Find the highest number for this type+year
   123→  const existing = await db.document.findMany({
   124→    where: {
   125→      type,
   126→      number: { startsWith: `${prefix}-${year}-` },
   127→    },
   128→    orderBy: { number: 'desc' },
   129→    take: 1,
   130→  })
   131→
   132→  let next = 1
   133→  if (existing.length > 0) {
   134→    const match = existing[0].number.match(/-(\d+)$/)
   135→    if (match) next = parseInt(match[1], 10) + 1
   136→  }
   137→
   138→  return `${prefix}-${year}-${String(next).padStart(4, '0')}`
   139→}

   143→/**
   144→ * Sanitize text for WinAnsi encoding (pdf-lib default Helvetica).
   145→ * Replaces Unicode characters that WinAnsi cannot encode with ASCII equivalents.
   146→ */
   147→function sanitizeText(text: string | undefined | null): string {
   148→  if (!text) return ''
   149→  return String(text)
   150→    // Narrow no-break space (U+202F) — used by fr-FR toLocaleString as thousands separator
   151→    .replace(/\u202F/g, ' ')
   152→    // No-break space (U+00A0)
   153→    .replace(/\u00A0/g, ' ')
   154→    // Thin space, hair space, etc.
   155→    .replace(/[\u2009\u200A\u200B\u200C\u200D\u2060]/g, '')
   156→    // Curly quotes → straight quotes
   157→    .replace(/[\u2018\u2019]/g, "'")
   158→    .replace(/[\u201C\u201D]/g, '"')
   159→    // Em/en dash → hyphen
   160→    .replace(/[\u2013\u2014]/g, '-')
   161→    // Ellipsis → ...
   162→    .replace(/\u2026/g, '...')
   163→    // Bullet → *
   164→    .replace(/[\u2022\u25CF]/g, '*')
   165→    // Copyright, trademark
   166→    .replace(/\u00A9/g, '(c)')
   167→    .replace(/\u00AE/g, '(R)')
   168→    .replace(/\u2122/g, '(TM)')
   169→    // Euro sign is OK in WinAnsi (0x80), keep it
   170→    // Strip any remaining non-WinAnsi characters
   171→    .replace(/[^\x20-\x7E\x80-\xFF]/g, '?')
   172→}

   174→/** Safe drawText wrapper that sanitizes text for WinAnsi encoding. */
   175→function drawTextSafe(
   176→  page: PDFPage,
   177→  text: string,
   178→  options: {
   179→    x: number
   180→    y: number
   181→    size: number
   182→    font: PDFFont
   183→    color?: ReturnType<typeof rgb>
   184→    opacity?: number
   185→    rotate?: ReturnType<typeof degrees>
   186→    align?: typeof TextAlignment
   187→  }
   188→): void {
   189→  const sanitized = sanitizeText(text)
   190→  // pdf-lib doesn't support align in drawText, we handle alignment manually for totals
   191→  if (options.align === TextAlignment.Right) {
   192→    const textWidth = options.font.widthOfTextAtSize(sanitized, options.size)
   193→    page.drawText(sanitized, {
   194→      ...options,
   195→      x: options.x - textWidth,
   196→    })
   197→  } else {
   198→    page.drawText(sanitized, options)
   199→  }
   200→}

   202→function formatMoney(amount: number, currency: string): string {
   203→  const symbols: Record<string, string> = { EUR: 'EUR', USD: 'USD', GBP: 'GBP', MAD: 'MAD', AED: 'AED', SAR: 'SAR' }
   204→  const symbol = symbols[currency] || currency
   205→  // Use 'en-US' formatting then replace comma with dot for FR style, avoids Unicode narrow no-break space (U+202F)
   206→  const formatted = Math.abs(amount).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })
   207→  const result = currency === 'EUR' ? `${formatted} ${symbol}` : currency === 'USD' || currency === 'GBP' ? `${symbol} ${formatted}` : `${formatted} ${symbol}`
   208→  return result
   209→}

   211→function formatDate(date: Date | undefined | null): string {
   212→  if (!date) return '-'
   213→  const d = new Date(date)
   214→  const day = String(d.getDate()).padStart(2, '0')
   215→  const month = String(d.getMonth() + 1).padStart(2, '0')
   216→  const year = d.getFullYear()
   217→  return `${day}/${month}/${year}`
   218→}

   341→function drawMetaBox(page: PDFPage, fontBold: PDFFont, fontRegular: PDFFont, data: DocumentData, startY: number) {
   342→  const { width } = page.getSize()
   343→  const boxY = startY - 10

   350→    width: width - 100,
   351→    height: boxHeight,
   352→    color: COLORS.veryLight,
   353→    borderColor: COLORS.border,
   354→    borderWidth: 1,
   355→  })
   356→
   357→  // Subject
   358→  page.drawText('Objet :', { x: 64, y: boxY - 16, size: 9, font: fontBold, color: COLORS.gray })
   359→  page.drawText(data.subject, { x: 110, y: boxY - 16, size: 10, font: fontBold, color: COLORS.dark })
   360→
   361→  // Issue date
   362→  page.drawText('Date émission :', { x: 64, y: boxY - 32, size: 9, font: fontBold, color: COLORS.gray })
   363→  page.drawText(formatDate(data.issueDate || new Date()), { x: 160, y: boxY - 32, size: 9, font: fontRegular, color: COLORS.dark })
   364→
   365→  // Due date / validity
   366→  const dueLabel = data.type === 'quote' ? 'Valable jusqu\'au :' : data.type === 'invoice' ? 'Échéance :' : 'Date :'
   367→  const dueValue = data.type === 'quote'
   368→    ? formatDate(new Date(Date.now() + 30 * 24 * 60 * 60 * 1000))
   369→    : formatDate(data.dueDate)
   370→  page.drawText(dueLabel, { x: 320, y: boxY - 32, size: 9, font: fontBold, color: COLORS.gray })
   371→  page.drawText(dueValue, { x: 410, y: boxY - 32, size: 9, font: fontRegular, color: COLORS.dark })
   372→
   373→  // Status
   374→  const statusLabel = data.type === 'receipt' ? 'Payé le :' : 'Statut :'
   375→  const statusValue = data.type === 'receipt'
   376→    ? formatDate(data.paidAt || new Date())
   377→    : data.type === 'invoice' && data.paidAt
   378→      ? `Payé le ${formatDate(data.paidAt)}`
   379→      : 'En attente'
   380→  page.drawText(statusLabel, { x: 64, y: boxY - 48, size: 9, font: fontBold, color: COLORS.gray })
   381→  page.drawText(statusValue, { x: 160, y: boxY - 48, size: 9, font: fontBold, color: data.paidAt ? COLORS.primary : COLORS.amber })
   382→
   383→  return boxY - boxHeight - 20
   384→}

   386→function drawItemsTable(page: PDFPage, fontBold: PDFFont, fontRegular: PDFFont, data: DocumentData, startY: number): number {
   387→  const { width } = page.getSize()
   388→  const tableX = 50
   389→  const tableWidth = width - 100
   390→  let yPos = startY
   391→
   392→  // Table header
   393→  page.drawRectangle({
   394→    x: tableX,
   395→    y: yPos - 22,
   396→    width: tableWidth,
   397→    height: 22,
   398→    color: COLORS.primary,
   399→  })
   400→
   401→  page.drawText('Description', { x: tableX + 12, y: yPos - 15, size: 9, font: fontBold, color: COLORS.white })
   402→  page.drawText('Qté', { x: tableX + tableWidth - 230, y: yPos - 15, size: 9, font: fontBold, color: COLORS.white })
   403→  page.drawText('Prix unitaire', { x: tableX + tableWidth - 170, y: yPos - 15, size: 9, font: fontBold, color: COLORS.white })
   404→  page.drawText('Total', { x: tableX + tableWidth - 70, y: yPos - 15, size: 9, font: fontBold, color: COLORS.white })
   405→
   406→  yPos -= 22
   407→
   408→  // Items rows
   409→  let altRow = false
   410→  let subtotal = 0
   411→  for (const item of data.items) {
   412→    const rowHeight = 28
   413→    if (altRow) {
   414→      page.drawRectangle({
   415→        x: tableX,
   416→        y: yPos - rowHeight,
   417→        width: tableWidth,
   418→        height: rowHeight,
   419→        color: COLORS.veryLight,
   420→      })
   421→    }
   422→    altRow = !altRow
   423→
   424→    // Description (truncated if too long)
   425→    const desc = item.description.length > 65 ? item.description.slice(0, 62) + '...' : item.description
   426→    page.drawText(desc, { x: tableX + 12, y: yPos - 18, size: 9, font: fontRegular, color: COLORS.dark })
   427→
   428→    page.drawText(String(item.quantity), { x: tableX + tableWidth - 230, y: yPos - 18, size: 9, font: fontRegular, color: COLORS.dark })
   429→    page.drawText(formatMoney(item.unitPrice, data.currency), { x: tableX + tableWidth - 170, y: yPos - 18, size: 9, font: fontRegular, color: COLORS.dark })
   430→    page.drawText(formatMoney(item.total, data.currency), { x: tableX + tableWidth - 70, y: yPos - 18, size: 9, font: fontBold, color: COLORS.dark })
   431→
   432→    subtotal += item.total
   433→    yPos -= rowHeight
   434→  }
   435→
   436→  // Border around table
   437→  page.drawRectangle({
   438→    x: tableX,
   439→    y: yPos,
   440→    width: tableWidth,
   441→    height: startY - yPos,
   442→    borderColor: COLORS.border,
   443→    borderWidth: 1,
   444→  })
   445→
   446→  // Totals
   447→  const taxRate = data.taxRate || 0
   448→  const taxAmount = subtotal * (taxRate / 100)
   449→  const total = subtotal + taxAmount
   450→
   451→  yPos -= 20
   452→  const totalsX = tableX + tableWidth - 240
   453→
   454→  // Subtotal
   455→  page.drawText('Sous-total :', { x: totalsX, y: yPos, size: 10, font: fontRegular, color: COLORS.gray })
   456→  page.drawText(formatMoney(subtotal, data.currency), { x: tableX + tableWidth - 12 - 90, y: yPos, size: 10, font: fontRegular, color: COLORS.dark, align: TextAlignment.Right })
   457→
   458→  yPos -= 18
   459→  if (taxRate > 0) {
   460→    page.drawText(`TVA (${taxRate}%) :`, { x: totalsX, y: yPos, size: 10, font: fontRegular, color: COLORS.gray })
   461→    page.drawText(formatMoney(taxAmount, data.currency), { x: tableX + tableWidth - 12 - 90, y: yPos, size: 10, font: fontRegular, color: COLORS.dark, align: TextAlignment.Right })
   462→    yPos -= 18
   463→  }
   464→
   465→  // Total (highlighted)
   466→  page.drawRectangle({
   467→    x: totalsX - 12,
   468→    y: yPos - 6,
   469→    width: tableX + tableWidth - totalsX,
   470→    height: 26,
   471→    color: COLORS.primaryLight,
   472→  })
   473→  page.drawText('TOTAL :', { x: totalsX, y: yPos, size: 12, font: fontBold, color: COLORS.primaryDark })
   474→  page.drawText(formatMoney(total, data.currency), { x: tableX + tableWidth - 12 - 90, y: yPos, size: 12, font: fontBold, color: COLORS.primaryDark, align: TextAlignment.Right })
   475→
   476→  return yPos - 40
   477→}

   539→async function buildPdf(rawData: DocumentData): Promise<Uint8Array> {
   540→  // Sanitize user-provided data for PDF metadata (setTitle, setSubject, etc.)
   541→  const data: DocumentData = {
   542→    ...rawData,
   543→    recipientName: sanitizeText(rawData.recipientName),
   544→    recipientEmail: sanitizeText(rawData.recipientEmail),
   545→    recipientCompany: sanitizeText(rawData.recipientCompany),
   546→    recipientAddress: sanitizeText(rawData.recipientAddress),
   547→    recipientCountry: sanitizeText(rawData.recipientCountry),
   548→    subject: sanitizeText(rawData.subject),
   549→    notes: sanitizeText(rawData.notes),
   550→    agreementTerms: sanitizeText(rawData.agreementTerms),
   551→    items: rawData.items.map((item) => ({
   552→      ...item,
   553→      description: sanitizeText(item.description),
   554→    })),
   555→  }
   556→
   557→  const pdfDoc = await PDFDocument.create()
   558→  const fontRegular = await pdfDoc.embedFont(StandardFonts.Helvetica)
   559→  const fontBold = await pdfDoc.embedFont(StandardFonts.HelveticaBold)
   560→  const fontItalic = await pdfDoc.embedFont(StandardFonts.HelveticaOblique)
   561→
   562→  // Embed HireNova logo (white variant — visible on colored header bands)
   563→  let logo: PDFImage | undefined
   564→  try {
   565→    logo = await embedHireNovaLogo(pdfDoc, 'white')
   566→  } catch (err) {
   567→    // Logo is optional — document can still be generated without it
   568→    console.warn('[documents] Logo not embedded:', err instanceof Error ? err.message : err)
   569→  }
   570→
   571→  pdfDoc.setTitle(sanitizeText(`${TYPE_META[data.type].label} ${data.number} - HireNova`))
   572→  pdfDoc.setAuthor(sanitizeText('HireNova - E-Society 2050'))
   573→  pdfDoc.setSubject(data.subject)
   574→  pdfDoc.setCreator('HireNova Document Engine')
   575→  pdfDoc.setProducer('HireNova Auto-Generator')
   576→  pdfDoc.setCreationDate(new Date())
   577→
   578→  const page = pdfDoc.addPage([595.28, 841.89]) // A4
   579→
   580→  // CRITICAL: Override drawText to auto-sanitize ALL text for WinAnsi encoding.
   581→  // This prevents "WinAnsi cannot encode" errors with Unicode chars (em dash, narrow no-break space, accented chars, etc.)
   582→  // Also handles right-alignment manually since pdf-lib's drawText doesn't support align natively.
   583→  const originalDrawText = page.drawText.bind(page)
   584→  ;(page as any).drawText = (text: string, options: any) => {
   585→    const sanitized = sanitizeText(text)
   586→    if (options.align === TextAlignment.Right) {
   587→      const textWidth = options.font.widthOfTextAtSize(sanitized, options.size)
   588→      const { align, ...rest } = options
   589→      return originalDrawText(sanitized, { ...rest, x: options.x - textWidth })
   590→    }
   591→    const { align, ...rest } = options
   592→    return originalDrawText(sanitized, rest)
   593→  }
   594→
   595→  const meta = TYPE_META[data.type]
   596→  let yPos = drawHeader(page, fontBold, fontRegular, data, meta.label, meta.color, logo)
   597→  yPos = drawMetaBox(page, fontBold, fontRegular, data, yPos)
   598→
   599→  // Content depends on document type
   600→  if (data.type === 'accounting_statement') {
   601→    // Bilan: custom layout (summary + invoice table + tax section)
   602→    yPos = drawBilanContent(page, fontBold, fontRegular, fontItalic, data, yPos)
   603→  } else if (data.type === 'agreement' && data.agreementTerms) {
   604→    // Agreement: contract clauses instead of items table
   605→    yPos = drawAgreementClauses(page, fontBold, fontRegular, fontItalic, data, yPos)
   606→  } else {
   607→    // Standard: items table
   608→    yPos = drawItemsTable(page, fontBold, fontRegular, data, yPos)
   609→  }
   610→
   611→  // Payment terms / acceptance block (not for bilan/receipt/credit_note)
   612→  if (data.type === 'quote') {
   613→    yPos = drawAcceptanceBlock(page, fontBold, fontRegular, yPos)
   614→  } else if (data.type === 'invoice') {
   615→    yPos = drawPaymentTerms(page, fontBold, fontRegular, data, yPos)
   616→  }
   617→
   618→  // ===== Electronic signature block (ALL document types) =====
   619→  // Placed before the footer, takes ~92pt at the bottom of the page.
   620→  if (data.signature) {
   621→    yPos = Math.min(yPos, 130) // ensure room for footer below signature
   622→    drawSignatureBlock(page, fontBold, fontRegular, fontItalic, data.signature, yPos)
   623→  }
   624→
   625→  drawFooter(page, fontBold, fontRegular, data)
   626→
   627→  const pdfBytes = await pdfDoc.save()
   628→  return pdfBytes
   629→}

   631→function drawAgreementClauses(
   632→  page: PDFPage,
   633→  fontBold: PDFFont,
   634→  fontRegular: PDFFont,
   635→  fontItalic: PDFFont,
   636→  data: DocumentData,
   637→  startY: number
   638→): number {
   639→  let yPos = startY
   640→
   641→  page.drawText('CLAUSES DU CONTRAT', { x: 50, y: yPos, size: 11, font: fontBold, color: COLORS.primaryDark })
   642→  yPos -= 20
   643→
   644→  const terms = data.agreementTerms || ''
   645→  const clauses = terms.split('\n\n').filter(Boolean)
   646→
   647→  for (let i = 0; i < clauses.length; i++) {
   648→    const clause = clauses[i]
   649→    const title = `Article ${i + 1}`
   650→    page.drawText(title, { x: 50, y: yPos, size: 10, font: fontBold, color: COLORS.dark })
   651→    yPos -= 14
   652→
   653→    // Wrap text
   654→    const maxWidth = 495
   655→    const words = clause.split(' ')
   656→    let line = ''
   657→    const lines: string[] = []
   658→    for (const word of words) {
   659→      const testLine = line ? line + ' ' + word : word
   660→      try {
   661→        const testWidth = fontRegular.widthOfTextAtSize(testLine, 9)
   662→        if (testWidth > maxWidth) {
   663→          if (line) lines.push(line)
   664→          line = word
   665→        } else {
   666→          line = testLine
   667→        }
   668→      } catch {
   669→        line = line ? line + ' ' + word : word
   670→      }
   671→    }
   672→    if (line) lines.push(line)
   673→
   674→    for (const ln of lines.slice(0, 6)) {
   675→      page.drawText(ln.slice(0, 100), { x: 60, y: yPos, size: 9, font: fontRegular, color: COLORS.gray })
   676→      yPos -= 12
   677→    }
   678→    yPos -= 8
   679→    if (yPos < 180) break
   680→  }
   681→
   682→  // Signatures
   683→  yPos = Math.max(yPos, 200)
   684→  page.drawText('Signatures', { x: 50, y: yPos, size: 11, font: fontBold, color: COLORS.dark })
   685→  yPos -= 20
   686→
   687→  page.drawLine({ start: { x: 50, y: yPos }, end: { x: 230, y: yPos }, thickness: 1, color: COLORS.gray })
   688→  page.drawLine({ start: { x: 320, y: yPos }, end: { x: 500, y: yPos }, thickness: 1, color: COLORS.gray })
   689→
   690→  page.drawText('Pour HireNova', { x: 50, y: yPos - 14, size: 9, font: fontBold, color: COLORS.dark })
   691→  page.drawText(data.recipientName, { x: 320, y: yPos - 14, size: 9, font: fontBold, color: COLORS.dark })
   692→
   693→  page.drawText('Date : _______________', { x: 50, y: yPos - 30, size: 9, font: fontRegular, color: COLORS.gray })
   694→  page.drawText('Date : _______________', { x: 320, y: yPos - 30, size: 9, font: fontRegular, color: COLORS.gray })
   695→
   696→  return yPos - 50
   697→}

   742→/**
   743→ * Generate a document PDF and persist it in the database.
   744→ * Returns the generated document with PDF base64.
   745→ */
   746→export async function generateDocument(data: DocumentData): Promise<GeneratedDocument> {
   747→  // Calculate totals
   748→  const subtotal = data.items.reduce((sum, item) => sum + item.total, 0)
   749→  const taxRate = data.taxRate || 0
   750→  const taxAmount = subtotal * (taxRate / 100)
   751→  const total = subtotal + taxAmount
   752→
   753→  // ===== Apply electronic signature (if not already provided) =====
   754→  // The signature hash is computed from the document fingerprint (number + parties + total + items).
   755→  // This makes every document tamper-evident: any change to the data invalidates the hash.
   756→  if (!data.signature) {
   757→    const fingerprint: SignatureFingerprint = {
   758→      number: data.number,
   759→      type: data.type,
   760→      issuer: data.issuerLegal || 'E-Society 2050',
   761→      recipient: data.recipientName,
   762→      recipientEmail: data.recipientEmail,
   763→      subject: data.subject,
   764→      items: JSON.stringify(data.items),
   765→      total,
   766→      currency: data.currency,
   767→      issueDate: (data.issueDate || new Date()).toISOString(),
   768→    }
   769→    data.signature = await applySignature(fingerprint)
   770→  }
   771→
   772→  // Build the PDF (now includes logo + signature block)
   773→  const pdfBytes = await buildPdf(data)
   774→  const pdfBase64 = Buffer.from(pdfBytes).toString('base64')
   775→
   776→  // Persist to DB (includes signature + bilan fields)
   777→  const doc = await db.document.create({
   778→    data: {
   779→      type: data.type,
   780→      number: data.number,
   781→      recipientName: data.recipientName,
   782→      recipientEmail: data.recipientEmail,
   783→      recipientCompany: data.recipientCompany || null,
   784→      recipientAddress: data.recipientAddress || null,
   785→      recipientCountry: data.recipientCountry || null,
   786→      issuerName: data.issuerName || 'HireNova',
   787→      issuerLegal: data.issuerLegal || 'E-Society 2050',
   788→      issuerAddress: data.issuerAddress || 'Casablanca, Maroc',
   789→      issuerEmail: data.issuerEmail || 'hello@hirenova.com',
   790→      issuerVAT: data.issuerVAT,
   791→      subject: data.subject,
   792→      items: JSON.stringify(data.items),
   793→      currency: data.currency,
   794→      subtotal,
   795→      taxRate,
   796→      taxAmount,
   797→      total,
   798→      status: data.type === 'accounting_statement' ? 'finalized' : (data.paidAt ? 'paid' : 'draft'),
   799→      issueDate: data.issueDate || new Date(),
   800→      dueDate: data.dueDate || null,
   801→      paidAt: data.paidAt || null,
   802→      userId: data.userId || null,
   803→      inquiryId: data.inquiryId || null,
   804→      // Signature fields
   805→      signatureHash: data.signature.hash,
   806→      signatureDate: data.signature.signedAt,
   807→      signedBy: data.signature.signedBy,
   808→      signatureSerial: data.signature.serial,
   809→      // Bilan-specific fields (only populated for accounting_statement type)
   810→      periodStart: data.periodStart || null,
   811→      periodEnd: data.periodEnd || null,
   812→      linkedDocIds: data.linkedInvoices ? JSON.stringify(data.linkedInvoices.map((inv) => inv.number)) : null,
   813→      invoiceCount: data.invoiceCount || 0,
   814→      platformFees: data.platformFees || 0,
   815→      royaltyFees: data.royaltyFees || 0,
   816→      netProfit: data.netProfit || 0,
   817→      totalCollected: data.totalCollected || total,
   818→      pdfBase64,
   819→      notes: data.notes || null,
   820→    },
   821→  })
   822→
   823→  return {
   824→    id: doc.id,
   825→    number: doc.number,
   826→    type: doc.type as DocumentType,
   827→    pdfBase64,
   828→    total,
   829→  }
   830→}

  1190→/**
  1191→ * Generate an accounting statement (bilan comptable) for a given period.
  1192→ *
  1193→ * Aggregates all paid invoices in the period, computes:
  1194→ *   - Total HT (subtotal)
  1195→ *   - Total TVA (tax)
  1196→ *   - Total TTC encaissé (totalCollected)
  1197→ *   - Frais de plateforme (payment processing + infra)
  1198→ *   - Redevances (third-party royalties)
  1199→ *   - Bénéfice net (totalCollected - platformFees - royaltyFees)
  1200→ *
  1201→ * The resulting bilan document is linked to all source invoices via linkedDocIds,
  1202→ * making it fully auditable for tax declaration and profit tracking.
  1203→ *
  1204→ * @param params.periodStart  Period start date (inclusive)
  1205→ * @param params.periodEnd    Period end date (inclusive)
  1206→ * @param params.platformFeesRate  Platform fee rate as decimal (e.g. 0.03 = 3%). Default 0.03
  1207→ * @param params.platformFeesFixed  Fixed platform fee per invoice (e.g. 0.30). Default 0.30
  1208→ * @param params.royaltyFees  Total royalties for the period. Default 0
  1209→ * @param params.currency     Currency code. Default EUR
  1210→ */
  1211→export async function generateAccountingStatement(params: {
  1212→  periodStart: Date
  1213→  periodEnd: Date
  1214→  platformFeesRate?: number
  1215→  platformFeesFixed?: number
  1216→  royaltyFees?: number
  1217→  currency?: string
  1218→  notes?: string
  1219→}): Promise<GeneratedDocument & { invoiceCount: number; netProfit: number; totalCollected: number }> {
  1220→  const currency = params.currency || 'EUR'
  1221→  const feesRate = params.platformFeesRate ?? 0.03   // 3% default (Stripe/LemonSqueezy-like)
  1222→  const feesFixed = params.platformFeesFixed ?? 0.30 // 0.30 per invoice
  1223→  const royaltyFees = params.royaltyFees ?? 0
  1224→
  1225→  // Query all paid invoices in the period
  1226→  const invoices = await db.document.findMany({
  1227→    where: {
  1228→      type: 'invoice',
  1229→      paidAt: {
  1230→        gte: params.periodStart,
  1231→        lte: params.periodEnd,
  1232→      },
  1233→    },
  1234→    orderBy: { paidAt: 'asc' },
  1235→  })
  1236→
  1237→  // Aggregate
  1238→  let subtotal = 0
  1239→  let taxAmount = 0
  1240→  let totalCollected = 0
  1241→  const linkedInvoices = invoices.map((inv) => {
  1242→    const invSubtotal = inv.subtotal
  1243→    const invTax = inv.taxAmount
  1244→    const invTotal = inv.total
  1245→    subtotal += invSubtotal
  1246→    taxAmount += invTax
  1247→    totalCollected += invTotal
  1248→    return {
  1249→      number: inv.number,
  1250→      date: inv.paidAt || inv.issueDate,
  1251→      client: inv.recipientCompany || inv.recipientName,
  1252→      subtotal: invSubtotal,
  1253→      taxAmount: invTax,
  1254→      total: invTotal,
  1255→      status: inv.status,
  1256→    }
  1257→  })
  1258→
  1259→  // Compute charges
  1260→  const platformFees = invoices.length * feesFixed + totalCollected * feesRate
  1261→  const netProfit = totalCollected - platformFees - royaltyFees
  1262→
  1263→  const number = await nextDocumentNumber('accounting_statement')
  1264→  const periodLabel = `${formatDate(params.periodStart)} - ${formatDate(params.periodEnd)}`
  1265→
  1266→  // Items stored for audit (one item per invoice)
  1267→  const items: DocumentItem[] = linkedInvoices.map((inv) => ({
  1268→    description: `${inv.number} - ${inv.client}`,
  1269→    quantity: 1,
  1270→    unitPrice: inv.total,
  1271→    total: inv.total,
  1272→  }))
  1273→
  1274→  const result = await generateDocument({
  1275→    type: 'accounting_statement',
  1276→    number,
  1277→    recipientName: 'HireNova — Direction Financière',
  1278→    recipientEmail: 'finance@hirenova.com',
  1279→    recipientCompany: 'E-Society 2050',
  1280→    recipientAddress: 'Casablanca, Maroc',
  1281→    subject: `Bilan comptable — ${periodLabel} — ${invoices.length} facture(s)`,
  1282→    items: items.length > 0 ? items : [{ description: 'Aucune facture sur la période', quantity: 0, unitPrice: 0, total: 0 }],
  1283→    currency,
  1284→    taxRate: taxAmount > 0 && subtotal > 0 ? (taxAmount / subtotal) * 100 : 0,
  1285→    periodStart: params.periodStart,
  1286→    periodEnd: params.periodEnd,
  1287→    linkedInvoices,
  1288→    invoiceCount: invoices.length,
  1289→    platformFees,
  1290→    royaltyFees,
  1291→    netProfit,
  1292→    totalCollected,
  1293→    notes: params.notes || `Bilan généré automatiquement. ${invoices.length} facture(s) payée(s) sur la période. Frais plateforme: ${(feesRate * 100).toFixed(1)}% + ${feesFixed.toFixed(2)} ${currency}/facture. Ce document lie toutes les factures payées pour le suivi fiscal (TVA, CA, bénéfice).`,
  1294→  })
  1295→
  1296→  return {
  1297→    ...result,
  1298→    invoiceCount: invoices.length,
  1299→    netProfit,
  1300→    totalCollected,
  1301→  }
  1302→}
