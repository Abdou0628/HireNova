   119→function StatCard({
   120→  icon: Icon,
   121→  label,
   122→  value,
   123→  sub,
   124→  trend,
   125→  color = 'emerald',
   126→}: {
   127→  icon: React.ElementType
   128→  label: string
   129→  value: string | number
   130→  sub?: string
   131→  trend?: string
   132→  color?: 'emerald' | 'amber' | 'blue' | 'rose'
   133→}) {
   134→  const colorMap = {
   135→    emerald: 'bg-emerald-50 text-emerald-600 border-emerald-200',
   136→    amber: 'bg-amber-50 text-amber-600 border-amber-200',
   137→    blue: 'bg-sky-50 text-sky-600 border-sky-200',
   138→    rose: 'bg-rose-50 text-rose-600 border-rose-200',
   139→  }
   140→  const iconBg = {
   141→    emerald: 'bg-emerald-100 text-emerald-700',
   142→    amber: 'bg-amber-100 text-amber-700',
   143→    blue: 'bg-sky-100 text-sky-700',
   144→    rose: 'bg-rose-100 text-rose-700',
   145→  }
   146→
   147→  return (
   148→    <Card className={`border ${colorMap[color]} transition-all hover:shadow-md`}>
   149→      <CardContent className="p-4 sm:p-5">
   150→        <div className="flex items-start justify-between">
   151→          <div className="space-y-2">
   152→            <p className="text-xs font-medium text-muted-foreground uppercase tracking-wider">{label}</p>
   153→            <p className="text-2xl sm:text-3xl font-bold text-foreground">{value}</p>
   154→            {sub && <p className="text-xs text-muted-foreground">{sub}</p>}
   155→            {trend && (
   156→              <div className="flex items-center gap-1 text-xs font-medium text-emerald-600">
   157→                <ArrowUpRight className="w-3 h-3" />
   158→                {trend}
   159→              </div>
   160→            )}
   161→          </div>
   162→          <div className={`p-2.5 rounded-xl ${iconBg[color]}`}>
   163→            <Icon className="w-5 h-5" />
   164→          </div>
   165→        </div>
   166→      </CardContent>
   167→    </Card>
   168→  )
   169→}

   171→function PlanBadge({ plan }: { plan: string }) {

   184→        Lifetime

   186→    )

   189→    <Badge variant="secondary" className="gap-1">
   190→      <UserX className="w-3 h-3" />

   192→    </Badge>
   193→  )

   196→function formatDate(dateStr: string) {
   197→  const d = new Date(dateStr)
   198→  return d.toLocaleDateString('fr-FR', {
   199→    day: '2-digit',
   200→    month: 'short',
   201→    year: 'numeric',
   202→    hour: '2-digit',
   203→    minute: '2-digit',
   204→  })
   205→}
   206→
   207→function formatShortDate(dateStr: string) {
   208→  const d = new Date(dateStr)
   209→  return d.toLocaleDateString('fr-FR', {
   210→    day: '2-digit',
   211→    month: 'short',
   212→  })
   213→}
