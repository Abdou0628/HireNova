    47→export default function CVForm() {
    48→  const {
    49→    formStep,
    50→    setFormStep,
    51→    language,
    52→    setLanguage,
    53→    template,
    54→    setTemplate,
    55→    formData,
    56→    updateFormData,
    57→    setStep,
    58→    setIsGenerating,

    62→
    63→  const [isSubmitting, setIsSubmitting] = useState(false)
    64→  const [phoneCountryCode, setPhoneCountryCode] = useState<string>('')
    65→
    66→  function handlePhoneCountryChange(code: string) {
    67→    setPhoneCountryCode(code)
    68→    const country = countries.find((c) => c.code === code)

    70→      // Remove any existing dial code from phone, prepend new one
    71→      const currentNumber = formData.phone.replace(/^[+\d\s\-()]+/, '').trim()

    73→    }
    74→  }

    77→    const country = countries.find((c) => c.code === phoneCountryCode)

    79→    const cleaned = number.replace(/[^\d\s]/g, '').trim()
    80→    updateFormData({ phone: dial ? `${dial} ${cleaned}` : cleaned })
    81→  }
    82→
    83→  function getPhoneNumberOnly(): string {
    84→    const country = countries.find((c) => c.code === phoneCountryCode)
    85→    const dial = country?.dial || ''
    86→    return formData.phone.startsWith(dial) ? formData.phone.slice(dial.length).trim() : formData.phone.replace(/^[+\d\s\-()]+/, '').trim()
    87→  }
    88→

    90→    enter: (direction: number) => ({ x: direction > 0 ? 300 : -300, opacity: 0 }),
    91→    center: { x: 0, opacity: 1 },

    93→  }
    94→

   104→        toast.error(t(language, 'errorFillFields'))
   105→        return false
   106→      }
   107→    }
   108→    return true
   109→  }
   110→

   313→                      </div>

   318→                        <Input
   319→                          id="fullName"
   320→                          value={formData.fullName}

   322→                          placeholder={language === 'fr' ? 'Jean Dupont' : language === 'en' ? 'John Smith' : 'محمد أحمد'}
