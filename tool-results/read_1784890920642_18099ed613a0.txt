    52→const personaFieldMap: Record<PersonaType, { labelKey: TranslationKey; phKey: TranslationKey }[]> = {

    67→
    68→export default function CVForm() {
    69→  const {
    70→    formStep,
    71→    setFormStep,
    72→    language,
    73→    setLanguage,
    74→    template,
    75→    setTemplate,
    76→    formData,
    77→    updateFormData,
    78→    setStep,
    79→    setIsGenerating,
    80→    setGeneratedCV,
    81→    setError,
    82→    setSelectedPersona,
    83→    selectedPersona,
    84→    updateCLFormData,
    85→    setGeneratedCL,
    86→    setIsCLGenerating,
    87→    setCLError,
    88→  } = useCVStore()
    89→
    90→  const [isSubmitting, setIsSubmitting] = useState(false)
    91→  const [phoneCountryCode, setPhoneCountryCode] = useState<string>('')
    92→  const [personaFields, setPersonaFields] = useState<Record<string, string>>({})
    93→
    94→  function handlePhoneCountryChange(code: string) {
    95→    setPhoneCountryCode(code)
    96→    const country = countries.find((c) => c.code === code)
    97→    if (country) {
    98→      // Remove any existing dial code from phone, prepend new one
    99→      const currentNumber = formData.phone.replace(/^[+\d\s\-()]+/, '').trim()
   100→      updateFormData({ phone: currentNumber ? `${country.dial} ${currentNumber}` : country.dial })
   101→    }
   102→  }
   103→
   104→  function handlePhoneNumberChange(number: string) {
   105→    const country = countries.find((c) => c.code === phoneCountryCode)
   106→    const dial = country?.dial || ''
   107→    const cleaned = number.replace(/[^\d\s]/g, '').trim()
   108→    updateFormData({ phone: dial ? `${dial} ${cleaned}` : cleaned })
   109→  }
   110→
   111→  function getPhoneNumberOnly(): string {
   112→    const country = countries.find((c) => c.code === phoneCountryCode)
   113→    const dial = country?.dial || ''
   114→    return formData.phone.startsWith(dial) ? formData.phone.slice(dial.length).trim() : formData.phone.replace(/^[+\d\s\-()]+/, '').trim()
   115→  }
   116→
   117→  const slideVariants = {
   118→    enter: (direction: number) => ({ x: direction > 0 ? 300 : -300, opacity: 0 }),
   119→    center: { x: 0, opacity: 1 },
   120→    exit: (direction: number) => ({ x: direction < 0 ? 300 : -300, opacity: 0 }),
   121→  }
   122→
   123→  function validateStep(step: number): boolean {
   124→    if (step === 0) {
   125→      if (!formData.fullName.trim() || !formData.email.trim()) {
   126→        toast.error(t(language, 'errorFillFields'))
   127→        return false
   128→      }
   129→    }
   130→    if (step === 1) {
   131→      if (!formData.targetJob.trim()) {
   132→        toast.error(t(language, 'errorFillFields'))
   133→        return false
   134→      }
   135→    }
   136→    return true
   137→  }
   138→
   139→  function handleNext() {
   140→    if (!validateStep(formStep)) return
   141→    if (formStep < 3) setFormStep(formStep + 1)
   142→  }
   143→
   144→  function handlePrevious() {
   145→    if (formStep > 0) setFormStep(formStep - 1)
   146→  }
   147→
   148→  async function handleGenerate() {
   149→    if (!validateStep(formStep)) return
   150→    setIsSubmitting(true)
   151→    setStep('generating')
   152→    setIsGenerating(true)
   153→    setError(null)
   154→    setIsCLGenerating(formData.companyName.trim().length > 0)
   155→    setCLError(null)
   156→
   157→    const hasCompany = formData.companyName.trim().length > 0
   158→
   159→    try {
   160→      // Generate CV
   161→      const res = await fetch('/api/generate-cv', {
   162→        method: 'POST',
   163→        headers: { 'Content-Type': 'application/json' },
   164→        body: JSON.stringify({
   165→          ...formData,
   166→          language,
   167→          persona: selectedPersona || undefined,
   168→          personaFields: Object.keys(personaFields).length > 0 ? personaFields : undefined,
   169→        }),
   170→      })
   171→
   172→      const data = await res.json()
   173→      if (!res.ok) {
   174→        if (data.code === 'LIMIT_REACHED') {
   175→          setStep('landing')
   176→          toast.error(data.error, { duration: 6000 })
   177→          return
   178→        }
   179→        throw new Error(data.error || 'Erreur lors de la génération')
   180→      }
   181→
   182→      const generatedCV = data.cv
   183→      setGeneratedCV(generatedCV)
   184→
   185→      // If company name provided, auto-generate cover letter simultaneously
   186→      if (hasCompany) {
   187→        try {
   188→          // Auto-sync personal info to CL form data
   189→          updateCLFormData({
   190→            fullName: formData.fullName,
   191→            email: formData.email,
   192→            phone: formData.phone,
   193→            address: formData.address,
   194→            location: formData.location,
   195→            companyName: formData.companyName,
   196→            hiringManager: formData.hiringManager,
   197→            jobTitle: formData.targetJob,
   198→            tone: formData.clTone,
   199→          })
   200→
   201→          const clRes = await fetch('/api/generate-cover-letter', {
   202→            method: 'POST',
   203→            headers: { 'Content-Type': 'application/json' },
   204→            body: JSON.stringify({
   205→              fullName: formData.fullName,
   206→              email: formData.email,
   207→              phone: formData.phone,
   208→              address: formData.address,
   209→              location: formData.location,
   210→              companyName: formData.companyName,
   211→              hiringManager: formData.hiringManager,
   212→              jobTitle: formData.targetJob,
   213→              tone: formData.clTone,
   214→              language,
   215→              generatedCVSummary: generatedCV.summary,
   216→              generatedCVExperience: generatedCV.experience,
   217→              generatedCVEducation: generatedCV.education,
   218→              generatedCVSkills: generatedCV.skills,
   219→              generatedCVLanguages: generatedCV.languages,
   220→            }),
   221→          })
   222→
   223→          const clData = await clRes.json()
   224→          if (clRes.ok && clData.letter) {
   225→            setGeneratedCL(clData.letter)
   226→          }
   227→        } catch {
   228→          // CL generation failed silently - CV is still valid
   229→          setIsCLGenerating(false)
   230→        }
   231→      }
   232→
   233→      setTimeout(() => {
   234→        setStep('preview')
   235→        setIsGenerating(false)
   236→        setIsCLGenerating(false)
   237→      }, 1000)
   238→    } catch (err) {
   239→      setError(err instanceof Error ? err.message : 'Erreur inconnue')
   240→      setIsGenerating(false)
   241→      setIsCLGenerating(false)
   242→      setStep('form')
   243→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue')
   244→    } finally {
   245→      setIsSubmitting(false)
   246→    }
   247→  }
   248→
   249→  const isLastStep = formStep === 3
   250→
   251→  return (
   252→    <div className="min-h-screen flex flex-col bg-background">
   253→      {/* Header */}
   254→      <header className="w-full px-4 sm:px-6 lg:px-8 py-4 border-b bg-white/80 backdrop-blur-sm sticky top-0 z-50">
   255→        <div className="max-w-3xl mx-auto flex items-center justify-between">
   256→          <button
   257→            onClick={() => { setStep('landing'); setSelectedPersona(null) }}
   258→            className="flex items-center gap-2 text-muted-foreground hover:text-foreground transition-colors"
   259→          >
   260→            <Image src="/hirenova-logo.png" alt="HireNova" width={32} height={32} className="rounded-lg" />
   261→            <span className="font-semibold text-foreground hidden sm:inline">{t(language, 'siteTitle')}</span>
   262→          </button>
   263→          {selectedPersona && (
   264→            <span className="flex items-center gap-1.5 text-xs font-medium bg-emerald-50 text-emerald-700 px-2.5 py-1 rounded-full border border-emerald-200">
   265→              <span>{personaEmoji[selectedPersona]}</span>
   266→              <span>{t(language, personaNameKey[selectedPersona])}</span>
   267→            </span>
   268→          )}
   269→          <div className="flex items-center gap-1 bg-muted rounded-lg p-1">
   270→            {languages.map((lang) => (
   271→              <button
   272→                key={lang.id}
   273→                onClick={() => setLanguage(lang.id)}
   274→                className={`px-3 py-1.5 rounded-md text-sm font-medium transition-all ${
   275→                  language === lang.id
   276→                    ? 'bg-white shadow-sm text-foreground'
   277→                    : 'text-muted-foreground hover:text-foreground'
   278→                }`}
   279→              >
   280→                <span className="mr-1">{lang.flag}</span>
   281→                <span className="hidden sm:inline">{lang.id.toUpperCase()}</span>
   282→              </button>
   283→            ))}
   284→          </div>
   285→        </div>
   286→      </header>
   287→
   288→      <main className="flex-1 py-8 px-4 sm:px-6">
   289→        <div className="max-w-2xl mx-auto">
   290→          {/* Progress */}
   291→          <div className="mb-8">
   292→            <div className="flex items-center justify-between mb-4">
   293→              {stepTitles.map((titleKey, i) => {
   294→                const Icon = stepIcons[i]
   295→                const isActive = i === formStep
   296→                const isCompleted = i < formStep
   297→                return (
   298→                  <button
   299→                    key={titleKey}
   300→                    onClick={() => {
   301→                      if (isCompleted || i === formStep) setFormStep(i)
   302→                    }}
   303→                    className={`flex flex-col items-center gap-1.5 flex-1 cursor-pointer group ${
   304→                      !isCompleted && i !== formStep ? 'opacity-40' : ''
   305→                    }`}
   306→                  >
   307→                    <div
   308→                      className={`w-10 h-10 rounded-full flex items-center justify-center transition-all ${
   309→                        isActive
   310→                          ? 'bg-emerald-600 text-white shadow-lg shadow-emerald-600/25'
   311→                          : isCompleted
   312→                            ? 'bg-emerald-100 text-emerald-700'
   313→                            : 'bg-muted text-muted-foreground'
   314→                      }`}
   315→                    >
   316→                      <Icon className="w-5 h-5" />
   317→                    </div>
   318→                    <span
   319→                      className={`text-xs font-medium text-center hidden sm:block ${
   320→                        isActive ? 'text-emerald-600' : 'text-muted-foreground'
   321→                      }`}
   322→                    >
   323→                      {t(language, titleKey)}
   324→                    </span>
   325→                  </button>
   326→                )
   327→              })}
   328→            </div>
   329→            {/* Progress bar */}
   330→            <div className="h-1.5 bg-muted rounded-full overflow-hidden">
   331→              <motion.div
   332→                className="h-full bg-emerald-500 rounded-full"
   333→                animate={{ width: `${((formStep + 1) / 4) * 100}%` }}
   334→                transition={{ duration: 0.4, ease: 'easeOut' }}
   335→              />
   336→            </div>
   337→          </div>
   338→
   339→          {/* Form Content */}
   340→          <div className="bg-white rounded-2xl border shadow-sm p-6 sm:p-8 min-h-[400px] relative overflow-hidden">
   341→            <AnimatePresence mode="wait" custom={1}>
   342→              <motion.div
   343→                key={formStep}
   344→                custom={1}
   345→                variants={slideVariants}
   346→                initial="enter"
   347→                animate="center"
   348→                exit="exit"
   349→                transition={{ duration: 0.3, ease: 'easeInOut' }}
   350→              >
   351→                <h2 className="text-xl font-bold text-foreground mb-6 flex items-center gap-2">
   352→                  {(() => {
   353→                    const Icon = stepIcons[formStep]
   354→                    return <Icon className="w-5 h-5 text-emerald-600" />
   355→                  })()}
   356→                  {t(language, stepTitles[formStep])}
   357→                </h2>
   358→
   359→                {/* Step 1: Personal Info */}
   360→                {formStep === 0 && (
   361→                  <div className="space-y-4">
   362→                    {/* Photo Upload */}
   363→                    <div>
   364→                      <Label>{t(language, 'photo')}</Label>
   365→                      <div className="mt-1.5">
   366→                        {formData.photo ? (
   367→                          <div className="relative inline-block">
   368→                            <img
   369→                              src={formData.photo}
   370→                              alt="Photo"
   371→                              className="w-24 h-24 rounded-full object-cover border-2 border-emerald-200"
   372→                            />
   373→                            <button
   374→                              type="button"
   375→                              onClick={() => updateFormData({ photo: '' })}
   376→                              className="absolute -top-1 -right-1 w-6 h-6 rounded-full bg-red-500 text-white flex items-center justify-center hover:bg-red-600 transition-colors cursor-pointer shadow-sm"
   377→                            >
   378→                              <X className="w-3 h-3" />
   379→                            </button>
   380→                          </div>
   381→                        ) : (
   382→                          <label className="flex flex-col items-center justify-center w-full h-32 border-2 border-dashed border-muted-foreground/25 rounded-xl cursor-pointer hover:border-emerald-400 hover:bg-emerald-50/50 transition-all group">
   383→                            <input
   384→                              type="file"
   385→                              accept="image/*"
   386→                              className="hidden"
   387→                              onChange={(e) => {
   388→                                const file = e.target.files?.[0]
   389→                                if (!file) return
   390→                                if (file.size > 5 * 1024 * 1024) {
   391→                                  toast.error(language === 'fr' ? 'Image trop volumineuse (max 5 Mo)' : language === 'en' ? 'Image too large (max 5 MB)' : 'الصورة كبيرة جداً (الحد الأقصى 5 ميغابايت)')
   392→                                  return
   393→                                }
   394→                                const reader = new FileReader()
   395→                                reader.onload = () => {
   396→                                  updateFormData({ photo: reader.result as string })
   397→                                }
   398→                                reader.readAsDataURL(file)
   399→                              }}
   400→                            />
   401→                            <Camera className="w-8 h-8 text-muted-foreground/40 group-hover:text-emerald-500 transition-colors" />
   402→                            <span className="text-xs text-muted-foreground/60 mt-1.5 group-hover:text-emerald-600 transition-colors">
   403→                              {t(language, 'photoPlaceholder')}
   404→                            </span>
   405→                          </label>
   406→                        )}
   407→                      </div>
   408→                      <div className="flex items-start gap-1.5 mt-2">
   409→                        <Info className="w-3.5 h-3.5 text-amber-500 shrink-0 mt-0.5" />
   410→                        <p className="text-xs text-muted-foreground">{t(language, 'photoNote')}</p>
   411→                      </div>
   412→                    </div>
   413→                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   414→                      <div className="sm:col-span-2">
   415→                        <Label htmlFor="fullName">{t(language, 'fullName')} *</Label>
   416→                        <Input
   417→                          id="fullName"
   418→                          value={formData.fullName}
   419→                          onChange={(e) => updateFormData({ fullName: e.target.value })}
   420→                          placeholder={language === 'fr' ? 'Jean Dupont' : language === 'en' ? 'John Smith' : 'محمد أحمد'}
   421→                          className="mt-1.5"
   422→                        />
   423→                      </div>
   424→                      <div>
   425→                        <Label htmlFor="email">{t(language, 'email')} *</Label>
   426→                        <Input
   427→                          id="email"
   428→                          type="email"
   429→                          value={formData.email}
   430→                          onChange={(e) => updateFormData({ email: e.target.value })}
   431→                          placeholder="jean@exemple.com"
   432→                          className="mt-1.5"
   433→                        />
   434→                      </div>
   435→                      <div className="sm:col-span-2">
   436→                        <Label>{t(language, 'phone')}</Label>
   437→                        <div className="mt-1.5 flex gap-2">
   438→                          <select
   439→                            value={phoneCountryCode}
   440→                            onChange={(e) => handlePhoneCountryChange(e.target.value)}
   441→                            className="h-9 w-[180px] shrink-0 rounded-md border border-input bg-background px-3 text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500 focus:ring-offset-1 appearance-none cursor-pointer"
   442→                            aria-label={t(language, 'phoneCountry')}
   443→                          >
   444→                            <option value="">{t(language, 'phoneCountryPlaceholder')}</option>
   445→                            {countries.map((c) => (
   446→                              <option key={c.code} value={c.code}>
   447→                                {c.flag} {c.dial} {c.name}
   448→                              </option>
   449→                            ))}
   450→                          </select>
   451→                          <Input
   452→                            type="tel"
   453→                            value={getPhoneNumberOnly()}
   454→                            onChange={(e) => handlePhoneNumberChange(e.target.value)}
   455→                            placeholder={language === 'fr' ? '600 000 000' : language === 'en' ? '600 000 000' : language === 'ar' ? '600 000 000' : '600 000 000'}
   456→                            className="flex-1"
   457→                          />
   458→                        </div>
   459→                      </div>
   460→                      <div>
   461→                        <Label htmlFor="address">{t(language, 'address')}</Label>
   462→                        <Input
   463→                          id="address"
   464→                          value={formData.address}
   465→                          onChange={(e) => updateFormData({ address: e.target.value })}
   466→                          placeholder={language === 'fr' ? '123 Rue Mohammed V' : language === 'en' ? '123 Main Street' : language === 'ar' ? 'شارع محمد الخامس 123' : 'Calle Ejemplo 123'}
   467→                          className="mt-1.5"
   468→                        />
   469→                      </div>
   470→                      <div>
   471→                        <Label htmlFor="location">{t(language, 'location')}</Label>
   472→                        <Input
   473→                          id="location"
   474→                          value={formData.location}
   475→                          onChange={(e) => updateFormData({ location: e.target.value })}
   476→                          placeholder={language === 'fr' ? 'Casablanca, Maroc' : language === 'en' ? 'London, UK' : language === 'ar' ? 'الدار البيضاء، المغرب' : 'Madrid, España'}
   477→                          className="mt-1.5"
   478→                        />
   479→                      </div>
   480→                      <div>
   481→                        <Label htmlFor="linkedin">LinkedIn</Label>
   482→                        <Input
   483→                          id="linkedin"
   484→                          value={formData.linkedin}
   485→                          onChange={(e) => updateFormData({ linkedin: e.target.value })}
   486→                          placeholder="linkedin.com/in/jean-dupont"
   487→                          className="mt-1.5"
   488→                        />
   489→                      </div>
   490→                      <div className="sm:col-span-2">
   491→                        <Label htmlFor="website">{t(language, 'website')}</Label>
   492→                        <Input
   493→                          id="website"
   494→                          value={formData.website}
   495→                          onChange={(e) => updateFormData({ website: e.target.value })}
   496→                          placeholder="www.monsite.com"
   497→                          className="mt-1.5"
   498→                        />
   499→                      </div>
   500→                    </div>
   501→
   502→                    {/* Birth Information */}
