   714→export async function POST(request: NextRequest) {
   715→  try {
   716→    const body = await request.json()
   717→    const { message = '', mode = 'advisor', conversationHistory = [] } = body
   718→
   719→    const safeMode = (['advisor', 'support', 'products'].includes(mode) ? mode : 'advisor') as Mode
   720→    const lang = detectLanguage(message)
   721→
   722→    if (!message || typeof message !== 'string') {
   723→      return NextResponse.json({ success: true, response: INVALID_MSG[lang], lang })
   724→    }
   725→
   726→    // 1) Try rule-based answer first (instant, no SDK cost)
   727→    const ruleAnswer = ruleBasedResponse(message, safeMode, lang)
   728→    if (ruleAnswer) {
   729→      return NextResponse.json({ success: true, response: ruleAnswer, source: 'rules', lang })
   730→    }
   731→
   732→    // 2) Fallback to LLM via ZAI SDK
   733→    try {
   734→      const zai = ZAI.create()
   735→      const systemPrompt = KB_PRODUCTS[lang] + MODE_PROMPTS[lang][safeMode]
   736→
   737→      const messages = [
   738→        { role: 'system' as const, content: systemPrompt },
   739→        ...conversationHistory.slice(-10).map((m: any) => ({ role: m.role as string, content: m.content })),
   740→        { role: 'user' as const, content: message }
   741→      ]
   742→
   743→      const res = await zai.chat.completions.create({
   744→        model: 'deepseek-chat',
   745→        messages,
   746→        temperature: 0.7,
   747→        max_tokens: 800
   748→      })
   749→
   750→      const response = res.choices?.[0]?.message?.content?.trim()
   751→      if (response) {
   752→        return NextResponse.json({ success: true, response, source: 'llm', lang })
   753→      }
   754→    } catch (sdkErr) {
   755→      console.error('[chatbot] SDK error:', sdkErr instanceof Error ? sdkErr.message : String(sdkErr))
   756→    }
   757→
   758→    // 3) Final graceful fallback
   759→    return NextResponse.json({ success: true, response: FALLBACK[lang], source: 'fallback', lang })
   760→  } catch (error) {
   761→    console.error('[chatbot] route error:', error instanceof Error ? error.message : String(error))
   762→    return NextResponse.json({ success: true, response: ERROR_MSG.fr, lang: 'fr' })
   763→  }
