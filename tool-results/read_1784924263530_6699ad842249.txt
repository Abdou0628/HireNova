   112→  async function handleImportCv(e: React.ChangeEvent<HTMLInputElement>) {
   113→    const file = e.target.files?.[0]
   114→    if (!file) return
   115→    setIsImportingCv(true)
   116→    try {
   117→      const formData = new FormData()
   118→      formData.append('file', file)
   119→      formData.append('language', language)
   120→      const res = await fetch('/api/import-cv', {
   121→        method: 'POST',
   122→        body: formData,
   123→      })
   124→      const data = await res.json()
   125→      if (!res.ok) throw new Error(data.error || 'Erreur lors de l\'import')
   126→      // Auto-fill form fields from parsed CV
   127→      if (data.fullName) updateFormData({ fullName: data.fullName })
   128→      if (data.email) updateFormData({ email: data.email })
   129→      if (data.phone) updateFormData({ phone: data.phone })
   130→      if (data.location) updateFormData({ location: data.location })
   131→      if (data.targetJob) updateFormData({ targetJob: data.targetJob })
   132→      if (data.industry) updateFormData({ industry: data.industry })
   133→      if (data.experience) updateFormData({ experience: data.experience })
   134→      if (data.education) updateFormData({ education: data.education })
   135→      if (data.skills) updateFormData({ skills: data.skills })
   136→      if (data.languages) updateFormData({ languages: data.languages })
   137→      if (data.summary) updateFormData({ summary: data.summary })
   138→      if (data.linkedin) updateFormData({ linkedin: data.linkedin })
   139→      if (data.website) updateFormData({ website: data.website })
   140→      toast.success(t(language, 'importCvSuccess'))
   141→    } catch (err) {
   142→      const message = err instanceof Error ? err.message : 'Erreur inconnue'
   143→      toast.error(message)
   144→    } finally {
   145→      setIsImportingCv(false)
   146→      if (fileInputRef.current) fileInputRef.current.value = ''
   147→    }
   148→  }
   149→
   150→  function handlePhoneCountryChange(code: string) {
   151→    setPhoneCountryCode(code)
   152→    const country = countries.find((c) => c.code === code)
   153→    if (country) {
   154→      // Remove any existing dial code from phone, prepend new one
   155→      const currentNumber = formData.phone.replace(/^[+\d\s\-()]+/, '').trim()
   156→      updateFormData({ phone: currentNumber ? `${country.dial} ${currentNumber}` : country.dial })
   157→    }
   158→  }
   159→
   160→  function handlePhoneNumberChange(number: string) {
   161→    const country = countries.find((c) => c.code === phoneCountryCode)
   162→    const dial = country?.dial || ''
   163→    const cleaned = number.replace(/[^\d\s]/g, '').trim()
   164→    updateFormData({ phone: dial ? `${dial} ${cleaned}` : cleaned })
   165→  }
   166→
   167→  function getPhoneNumberOnly(): string {
   168→    const country = countries.find((c) => c.code === phoneCountryCode)
   169→    const dial = country?.dial || ''
   170→    return formData.phone.startsWith(dial) ? formData.phone.slice(dial.length).trim() : formData.phone.replace(/^[+\d\s\-()]+/, '').trim()
   171→  }

   173→  const slideVariants = {

   176→    exit: (direction: number) => ({ x: direction < 0 ? 300 : -300, opacity: 0 }),
   177→  }

   179→  function validateStep(step: number): boolean {

   181→      if (!formData.fullName.trim() || !formData.email.trim()) {
   182→        toast.error(t(language, 'errorFillFields'))
   183→        return false
   184→      }
   185→    }
   186→    if (step === 1) {
   187→      if (!formData.targetJob.trim()) {
   188→        toast.error(t(language, 'errorFillFields'))
   189→        return false
   190→      }
   191→    }
   192→    return true
   193→  }
   194→
   195→  function handleNext() {
   196→    if (!validateStep(formStep)) return

   200→  function handlePrevious() {

   218→        method: 'POST',

   255→            tone: formData.clTone,
   256→          })
   257→
   258→          const clRes = await fetch('/api/generate-cover-letter', {
   259→            method: 'POST',
   260→            headers: { 'Content-Type': 'application/json' },
   261→            body: JSON.stringify({
   262→              fullName: formData.fullName,
   263→              email: formData.email,
   264→              phone: formData.phone,
   265→              address: formData.address,
   266→              location: formData.location,
   267→              companyName: formData.companyName,
   268→              hiringManager: formData.hiringManager,
   269→              jobTitle: formData.targetJob,
   270→              tone: formData.clTone,

   298→      setIsCLGenerating(false)
   299→      setStep('form')
   300→      toast.error(err instanceof Error ? err.message : 'Erreur inconnue')
   301→    } finally {
   302→      setIsSubmitting(false)
   303→    }
   304→  }

   311→      <header className="w-full px-4 sm:px-6 lg:px-8 py-4 border-b bg-white/80 backdrop-blur-sm sticky top-0 z-50">
   312→        <div className="max-w-3xl mx-auto flex items-center justify-between">
   313→          <button
   314→            onClick={() => { setStep('landing'); setSelectedPersona(null) }}
   315→            className="flex items-center gap-2 text-muted-foreground hover:text-foreground transition-colors"
   316→          >
   317→            <div className="flex flex-col items-center">
   318→              <Image src="/hirenova-logo.png" alt="HireNova" width={32} height={32} className="rounded-lg" />
   319→              <span className="text-[9px] font-semibold text-emerald-600 tracking-wide">POWERED BY IA</span>
   320→            </div>
   321→            <span className="font-semibold text-foreground hidden sm:inline">{t(language, 'siteTitle')}</span>
   322→          </button>

   324→            <span className="flex items-center gap-1.5 text-xs font-medium bg-emerald-50 text-emerald-700 px-2.5 py-1 rounded-full border border-emerald-200">
   325→              <span>{personaEmoji[selectedPersona]}</span>
   326→              <span>{t(language, personaNameKey[selectedPersona])}</span>
   327→            </span>
   328→          )}
   329→          <div className="flex items-center gap-1 bg-muted rounded-lg p-1">
   330→            {languages.map((lang) => (
   331→              <button
   332→                key={lang.id}
   333→                onClick={() => setLanguage(lang.id)}
   334→                className={`px-3 py-1.5 rounded-md text-sm font-medium transition-all ${
   335→                  language === lang.id
   336→                    ? 'bg-white shadow-sm text-foreground'
   337→                    : 'text-muted-foreground hover:text-foreground'
   338→                }`}
   339→              >
   340→                <span className="mr-1">{lang.flag}</span>

   364→                      !isCompleted && i !== formStep ? 'opacity-40' : ''
   365→                    }`}
   366→                  >
   367→                    <div
   368→                      className={`w-10 h-10 rounded-full flex items-center justify-center transition-all ${
   369→                        isActive
   370→                          ? 'bg-emerald-600 text-white shadow-lg shadow-emerald-600/25'
   371→                          : isCompleted
   372→                            ? 'bg-emerald-100 text-emerald-700'

   399→          {/* Form Content */}

   401→            <AnimatePresence mode="wait" custom={1}>
   402→              <motion.div
   403→                key={formStep}
   404→                custom={1}
   405→                variants={slideVariants}
   406→                initial="enter"
   407→                animate="center"
   408→                exit="exit"
   409→                transition={{ duration: 0.3, ease: 'easeInOut' }}
   410→              >
   411→                <h2 className="text-xl font-bold text-foreground mb-6 flex items-center gap-2">
   412→                  {(() => {
   413→                    const Icon = stepIcons[formStep]
   414→                    return <Icon className="w-5 h-5 text-emerald-600" />
   415→                  })()}
   416→                  {t(language, stepTitles[formStep])}

   418→
   419→                {/* Step 1: Personal Info */}
   420→                {formStep === 0 && (

   434→                              variant="outline"

   436→                              onClick={() => fileInputRef.current?.click()}

   460→                    {/* Photo Upload */}
